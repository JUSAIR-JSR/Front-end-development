SQL Queries & Operations

WHERE Conditions

Filter data with conditions

-- Basic WHERE conditions
SELECT * FROM students WHERE age > 20;
SELECT * FROM students WHERE city = 'London';
SELECT * FROM students WHERE age BETWEEN 18 AND 25;

-- AND, OR, NOT operators
SELECT * FROM students WHERE age > 20 AND city = 'New York';
SELECT * FROM students WHERE age < 18 OR city = 'Tokyo';
SELECT * FROM students WHERE NOT age = 22;

-- IN operator
SELECT * FROM students WHERE city IN ('New York', 'London', 'Paris');
SELECT * FROM students WHERE age IN (20, 22, 25);

-- IS NULL / IS NOT NULL
SELECT * FROM students WHERE email IS NULL;
SELECT * FROM students WHERE email IS NOT NULL;

-- Comparison operators
SELECT * FROM students WHERE age = 20;    -- Equal
SELECT * FROM students WHERE age != 20;   -- Not equal
SELECT * FROM students WHERE age <> 20;   -- Not equal (alternative)
SELECT * FROM students WHERE age > 20;    -- Greater than
SELECT * FROM students WHERE age < 20;    -- Less than
SELECT * FROM students WHERE age >= 20;   -- Greater than or equal
SELECT * FROM students WHERE age <= 20;   -- Less than or equal

LIKE & Pattern Matching

Search for patterns in text data

-- LIKE operator with wildcards
SELECT * FROM students WHERE name LIKE 'A%';   -- Starts with A
SELECT * FROM students WHERE name LIKE '%n';   -- Ends with n
SELECT * FROM students WHERE name LIKE '%oh%'; -- Contains 'oh'
SELECT * FROM students WHERE name LIKE 'J_n';  -- J followed by any char, then n

-- NOT LIKE
SELECT * FROM students WHERE name NOT LIKE 'A%';

-- Case sensitivity (MySQL is case-insensitive by default)
SELECT * FROM students WHERE name LIKE BINARY 'alice%'; -- Case sensitive

-- Multiple pattern matching
SELECT * FROM students WHERE name LIKE 'A%' OR name LIKE 'B%';

-- Using wildcards in the middle
SELECT * FROM students WHERE name LIKE 'A%e'; -- Starts with A, ends with e

-- ESCAPE character for literal %
SELECT * FROM products WHERE description LIKE '100\% off';

-- REGEXP (Regular Expressions)
SELECT * FROM students WHERE name REGEXP '^A';     -- Starts with A
SELECT * FROM students WHERE name REGEXP 'n$';     -- Ends with n
SELECT * FROM students WHERE name REGEXP 'John|Jane'; -- John or Jane
SELECT * FROM students WHERE name REGEXP '^[A-C]'; -- Starts with A, B, or C

-- Practical examples
-- Find students with email from specific domain
SELECT * FROM students WHERE email LIKE '%@gmail.com';

-- Find courses with specific words in title
SELECT * FROM courses WHERE title LIKE '%Database%' OR title LIKE '%Web%';

-- Find names with exactly 5 characters
SELECT * FROM students WHERE name LIKE '_____';

ORDER BY & LIMIT

Sort and limit query results

-- ORDER BY single column
SELECT * FROM students ORDER BY name ASC;   -- Ascending (A-Z)
SELECT * FROM students ORDER BY age DESC;   -- Descending (high to low)

-- ORDER BY multiple columns
SELECT * FROM students ORDER BY city ASC, name ASC;
SELECT * FROM students ORDER BY age DESC, name ASC;

-- ORDER BY with expressions
SELECT * FROM students ORDER BY LENGTH(name) DESC;

-- LIMIT results
SELECT * FROM students LIMIT 5;                    -- First 5 records
SELECT * FROM students LIMIT 10 OFFSET 5;         -- Skip 5, take 10
SELECT * FROM students ORDER BY id DESC LIMIT 3;  -- Last 3 records

-- Combining WHERE, ORDER BY, and LIMIT
SELECT name, age, city 
FROM students 
WHERE age > 20 
ORDER BY age DESC 
LIMIT 5;

-- Pagination example (page 2, 10 records per page)
SELECT * FROM students 
ORDER BY id 
LIMIT 10 OFFSET 10;

-- Top 3 oldest students
SELECT name, age 
FROM students 
ORDER BY age DESC 
LIMIT 3;

-- Random sampling
SELECT * FROM students 
ORDER BY RAND() 
LIMIT 5;

-- Finding minimum and maximum with LIMIT
-- Youngest student
SELECT * FROM students ORDER BY age ASC LIMIT 1;

-- Oldest student
SELECT * FROM students ORDER BY age DESC LIMIT 1;

-- Using LIMIT with UPDATE (MySQL specific)
UPDATE students SET city = 'Unknown' 
WHERE email IS NULL

-- Using LIMIT with DELETE

WHERE age < 18 
LIMIT 5;

Aggregate Functions

Perform calculations on data sets

-- Basic aggregate functions
SELECT COUNT(*) AS total_students FROM students;
SELECT COUNT(email) AS students_with_email FROM students; -- Non-null counts
SELECT COUNT(DISTINCT city) AS unique_cities FROM students;

SELECT MAX(age) AS max_age FROM students;
SELECT MIN(age) AS min_age FROM students;
SELECT AVG(age) AS average_age FROM students;
SELECT SUM(age) AS total_years FROM students;

-- Multiple aggregates in one query
SELECT 
    COUNT(*) AS total_students,
    AVG(age) AS average_age,
    MIN(age) AS youngest,
    MAX(age) AS oldest,
    SUM(age) AS total_years
FROM students;

-- Aggregates with WHERE clause
SELECT 
    COUNT(*) AS adult_students,
    AVG(age) AS avg_adult_age
FROM students 
WHERE age >= 18;

-- COUNT with DISTINCT
SELECT COUNT(DISTINCT city) AS cities_count FROM students;

-- String aggregates (MySQL 8.0+)
SELECT GROUP_CONCAT(name) AS all_names FROM students;
SELECT GROUP_CONCAT(DISTINCT city ORDER BY city) AS all_cities FROM students;

-- Statistical functions
SELECT 
    STDDEV(age) AS age_stddev,
    VARIANCE(age) AS age_variance
FROM students;

-- Conditional aggregates
SELECT 
    COUNT(CASE WHEN age < 20 THEN 1 END) AS under_20,
    COUNT(CASE WHEN age >= 20 THEN 1 END) AS over_20
FROM students;

-- Aggregates with expressions
SELECT 
    AVG(age * 365) AS avg_age_in_days,
    MAX(LENGTH(name)) AS longest_name_length
FROM students;

-- Practical examples
-- Student statistics by city
SELECT 
    city,
    COUNT(*) AS student_count,
    AVG(age) AS average_age,
    MIN(age) AS youngest,
    MAX(age) AS oldest
FROM students
GROUP BY city;

-- Course enrollment statistics
SELECT 
    COUNT(*) AS total_courses,
    AVG(duration_hours) AS avg_duration,
    SUM(duration_hours) AS total_hours
FROM courses
WHERE is_active = TRUE;

Advanced MySQL

GROUP BY & HAVING

Group data and filter groups

-- Create enrollments table for examples
CREATE TABLE enrollments (
    id INT PRIMARY KEY AUTO_INCREMENT,
    student_id INT,
    course_id INT,
    enrolled_date DATE DEFAULT CURRENT_DATE,
    grade CHAR(1),
    FOREIGN KEY (student_id) REFERENCES students(id),
    FOREIGN KEY (course_id) REFERENCES courses(id)
);

-- Basic GROUP BY
SELECT city, COUNT(*) AS student_count
FROM students
GROUP BY city;

-- GROUP BY with multiple columns
SELECT city, age, COUNT(*) AS count
FROM students
GROUP BY city, age;

-- GROUP BY with aggregate functions
SELECT 
    city,
    COUNT(*) AS total_students,
    AVG(age) AS average_age,
    MAX(age) AS max_age,
    MIN(age) AS min_age
FROM students
GROUP BY city;

-- HAVING clause (filter groups)
SELECT city, COUNT(*) AS student_count
FROM students
GROUP BY city
HAVING COUNT(*) > 1;

SELECT city, AVG(age) AS avg_age
FROM students
GROUP BY city
HAVING AVG(age) > 21;

-- GROUP BY with WHERE and HAVING
SELECT city, COUNT(*) AS student_count
FROM students
WHERE age >= 18
GROUP BY city
HAVING COUNT(*) >= 2;

-- GROUP BY with ORDER BY
SELECT city, COUNT(*) AS student_count
FROM students
GROUP BY city
ORDER BY student_count DESC;

-- Complex GROUP BY example
SELECT 
    EXTRACT(YEAR FROM enrolled_date) AS year,
    EXTRACT(MONTH FROM enrolled_date) AS month,
    COUNT(*) AS enrollments_count
FROM enrollments
GROUP BY year, month
HAVING COUNT(*) > 0
ORDER BY year DESC, month DESC;

-- ROLLUP extension (subtotals)
SELECT 
    COALESCE(city, 'All Cities') AS city,
    COUNT(*) AS student_count
FROM students
GROUP BY city WITH ROLLUP;

-- Multiple aggregates with HAVING
SELECT 
    city,
    COUNT(*) AS total,
    SUM(CASE WHEN age < 20 THEN 1 ELSE 0 END) AS under_20,
    SUM(CASE WHEN age >= 20 THEN 1 ELSE 0 END) AS over_20
FROM students
GROUP BY city
HAVING total > 1 AND over_20 > 0;

JOIN Operations

Combine data from multiple tables

-- Sample data for joins
INSERT INTO enrollments (student_id, course_id, grade) VALUES
(1, 1, 'A'), (1, 2, 'B'), (2, 1, 'C'), (3, 3, 'A');

-- INNER JOIN (most common)
SELECT 
    s.name AS student_name,
    c.title AS course_title,
    e.grade
FROM enrollments e
INNER JOIN students s ON e.student_id = s.id
INNER JOIN courses c ON e.course_id = c.id;

-- LEFT JOIN (all students, even without enrollments)
SELECT 
    s.name AS student_name,
    c.title AS course_title
FROM students s
LEFT JOIN enrollments e ON s.id = e.student_id
LEFT JOIN courses c ON e.course_id = c.id;

-- RIGHT JOIN (all courses, even without students)
SELECT 
    c.title AS course_title,
    s.name AS student_name
FROM courses c
LEFT JOIN enrollments e ON c.id = e.course_id
LEFT JOIN students s ON e.student_id = s.id;

-- Multiple JOIN conditions
SELECT 
    s.name,
    c.title,
    e.enrolled_date
FROM students s
JOIN enrollments e ON s.id = e.student_id
JOIN courses c ON e.course_id = c.id AND c.is_active = TRUE;

-- JOIN with WHERE clause
SELECT 
    s.name,
    c.title
FROM students s
JOIN enrollments e ON s.id = e.student_id
JOIN courses c ON e.course_id = c.id
WHERE s.age > 20 AND c.duration_hours > 40;

-- JOIN with aggregate functions
SELECT 
    c.title AS course_title,
    COUNT(e.student_id) AS enrollment_count,
    AVG(s.age) AS average_student_age
FROM courses c
LEFT JOIN enrollments e ON c.id = e.course_id
LEFT JOIN students s ON e.student_id = s.id
GROUP BY c.id, c.title;

-- Self JOIN (join table with itself)
CREATE TABLE employees (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100),
    manager_id INT,
    FOREIGN KEY (manager_id) REFERENCES employees(id)
);

SELECT 
    e.name AS employee,
    m.name AS manager
FROM employees e
LEFT JOIN employees m ON e.manager_id = m.id;

-- CROSS JOIN (Cartesian product)
SELECT s.name, c.title
FROM students s
CROSS JOIN courses c
LIMIT 10; -- Be careful with large tables!

-- NATURAL JOIN (joins on same column names)
-- Not recommended - can be unpredictable
SELECT * FROM students NATURAL JOIN enrollments;

-- USING clause (when join columns have same name)
SELECT *
FROM students s
JOIN enrollments e USING (id); -- If both have 'id' column

-- Complex JOIN with multiple conditions
SELECT 
    s.name AS student_name,
    c.title AS course_title,
    e.grade,
    CASE 
        WHEN e.grade IN ('A', 'B') THEN 'Good'
        WHEN e.grade = 'C' THEN 'Average'
        ELSE 'Needs Improvement'
    END AS performance
FROM students s
JOIN enrollments e ON s.id = e.student_id
JOIN courses c ON e.course_id = c.id
WHERE c.is_active = TRUE
ORDER BY s.name, c.title;

Subqueries

Nested queries for complex operations

-- Subquery in WHERE clause
SELECT name, age
FROM students
WHERE age = (SELECT MAX(age) FROM students);

SELECT name
FROM students
WHERE id IN (SELECT student_id FROM enrollments WHERE grade = 'A');

-- Subquery in SELECT clause
SELECT 
    name,
    age,
    (SELECT AVG(age) FROM students) AS average_age,
    age - (SELECT AVG(age) FROM students) AS age_difference
FROM students;

-- Correlated subquery
SELECT 
    s.name,
    s.city,
    (SELECT COUNT(*) FROM students s2 WHERE s2.city = s.city) AS city_count
FROM students s;

-- EXISTS subquery
SELECT name
FROM students s
WHERE EXISTS (
    SELECT 1 FROM enrollments e 
    WHERE e.student_id = s.id AND e.grade = 'A'
);

-- NOT EXISTS
SELECT name
FROM students s
WHERE NOT EXISTS (
    SELECT 1 FROM enrollments e WHERE e.student_id = s.id
);

-- Subquery in FROM clause (derived table)
SELECT 
    city,
    student_count
FROM (
    SELECT 
        city, 
        COUNT(*) AS student_count
    FROM students
    GROUP BY city
) AS city_stats
WHERE student_count > 1;

-- Multiple subqueries
SELECT 
    name,
    (SELECT COUNT(*) FROM enrollments WHERE student_id = students.id) AS course_count,
    (SELECT AVG(age) FROM students) AS overall_avg_age
FROM students
WHERE age > (SELECT AVG(age) FROM students);

-- Subquery with JOIN
SELECT 
    s.name,
    c.title
FROM students s
JOIN enrollments e ON s.id = e.student_id
JOIN courses c ON e.course_id = c.id
WHERE e.grade = (
    SELECT MAX(grade) 
    FROM enrollments e2 
    WHERE e2.student_id = s.id
);

-- ANY and ALL operators
SELECT name
FROM students
WHERE age > ANY (SELECT age FROM students WHERE city = 'New York');

SELECT name
FROM students
WHERE age > ALL (SELECT age FROM students WHERE city = 'Tokyo');

-- Practical examples
-- Students enrolled in more than 2 courses
SELECT name
FROM students
WHERE (
    SELECT COUNT(*) 
    FROM enrollments 
    WHERE student_id = students.id
) > 2;

-- Courses with above average enrollment
SELECT title
FROM courses
WHERE (
    SELECT COUNT(*) 
    FROM enrollments 
    WHERE course_id = courses.id
) > (SELECT AVG(course_count) FROM (
    SELECT COUNT(*) AS course_count 
    FROM enrollments 
    GROUP BY course_id
) AS counts);

Indexes & Performance

Optimize query performance with indexes

-- Create basic indexes
CREATE INDEX idx_students_email ON students(email);
CREATE INDEX idx_students_city ON students(city);
CREATE INDEX idx_students_age ON students(age);

-- Composite index
CREATE INDEX idx_students_city_age ON students(city, age);

-- Unique index
CREATE UNIQUE INDEX idx_unique_student_email ON students(email);

-- Full-text index (for text search)
CREATE TABLE articles (
    id INT PRIMARY KEY AUTO_INCREMENT,
    title VARCHAR(200),
    content TEXT,
    FULLTEXT(title, content)
);

-- Use full-text search
SELECT * FROM articles 
WHERE MATCH(title, content) AGAINST('database performance');

-- Show indexes
SHOW INDEX FROM students;

-- Drop index
DROP INDEX idx_students_city ON students;

-- EXPLAIN to analyze query performance
EXPLAIN SELECT * FROM students WHERE city = 'London';
EXPLAIN FORMAT=JSON SELECT * FROM students WHERE age > 20;

-- Force index usage
SELECT * FROM students FORCE INDEX (idx_students_city) WHERE city = 'London';

-- Analyze table for better performance
ANALYZE TABLE students;

-- Check table status
SHOW TABLE STATUS LIKE 'students';

-- Index on expression (MySQL 8.0+)
CREATE INDEX idx_name_length ON students((LENGTH(name)));

-- Partial index (MySQL 8.0+)
CREATE INDEX idx_active_students ON students(name) WHERE age >= 18;

-- Performance tips in practice
-- 1. Index columns used in WHERE, JOIN, ORDER BY
-- 2. Use covering indexes when possible
-- 3. Avoid SELECT * - specify only needed columns
-- 4. Use LIMIT to restrict result sets
-- 5. Normalize your database design

-- Query optimization examples
-- Slow query (without proper indexes)
SELECT * FROM students WHERE city = 'London' AND age > 20;

-- Optimized query (with composite index)
CREATE INDEX idx_city_age ON students(city, age);
SELECT name, email FROM students WHERE city = 'London' AND age > 20;

-- Using covering index
CREATE INDEX idx_covering ON students(city, age, name, email);
SELECT name, email FROM students WHERE city = 'London' AND age > 20;

Database Design & Advanced Features

Constraints & Keys

Ensure data integrity with constraints

-- Primary Key constraints
CREATE TABLE departments (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL UNIQUE
);

-- Foreign Key constraints
CREATE TABLE employees (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    dept_id INT,
    salary DECIMAL(10,2) CHECK (salary > 0),
    FOREIGN KEY (dept_id) REFERENCES departments(id)
        ON DELETE SET NULL
        ON UPDATE CASCADE
);

-- Check constraints (MySQL 8.0.16+)
CREATE TABLE products (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    price DECIMAL(10,2) CHECK (price > 0),
    category ENUM('Electronics', 'Books', 'Clothing') NOT NULL,
    stock_quantity INT CHECK (stock_quantity >= 0)
);

-- Default constraints
CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    is_active BOOLEAN DEFAULT TRUE,
    login_count INT DEFAULT 0
);

-- Not Null constraints
CREATE TABLE orders (
    id INT PRIMARY KEY AUTO_INCREMENT,
    customer_name VARCHAR(100) NOT NULL,
    order_date DATE NOT NULL,
    total_amount DECIMAL(10,2) NOT NULL
);

-- Unique constraints
CREATE TABLE employees (
    id INT PRIMARY KEY AUTO_INCREMENT,
    employee_code VARCHAR(20) UNIQUE NOT NULL,
    national_id VARCHAR(20) UNIQUE,
    email VARCHAR(100) UNIQUE NOT NULL
);

-- Composite primary key
CREATE TABLE order_items (
    order_id INT,
    product_id INT,
    quantity INT NOT NULL CHECK (quantity > 0),
    unit_price DECIMAL(10,2) NOT NULL CHECK (unit_price > 0),
    PRIMARY KEY (order_id, product_id),
    FOREIGN KEY (order_id) REFERENCES orders(id),
    FOREIGN KEY (product_id) REFERENCES products(id)
);

-- Adding constraints to existing tables
ALTER TABLE students 
ADD CONSTRAINT chk_age_positive CHECK (age > 0);

ALTER TABLE students 
ADD CONSTRAINT uk_student_email UNIQUE (email);

-- Dropping constraints
ALTER TABLE students DROP CHECK chk_age_positive;
ALTER TABLE students DROP INDEX uk_student_email;

-- Foreign key options
CREATE TABLE parent_child (
    parent_id INT,
    child_id INT,
    FOREIGN KEY (parent_id) REFERENCES parents(id)
        ON DELETE CASCADE    -- Delete children when parent deleted
        ON UPDATE CASCADE,   -- Update children when parent updated
    FOREIGN KEY (child_id) REFERENCES children(id)
        ON DELETE RESTRICT  -- Prevent deletion if children exist
        ON UPDATE CASCADE
);

-- Check existing constraints

    TABLE_NAME,
    CONSTRAINT_NAME,
    CONSTRAINT_TYPE
FROM information_schema.TABLE_CONSTRAINTS 
WHERE TABLE_SCHEMA = 'school';

-- Practical constraint examples
CREATE TABLE bank_accounts (
    account_number VARCHAR(20) PRIMARY KEY,
    account_holder VARCHAR(100) NOT NULL,
    balance DECIMAL(15,2) DEFAULT 0.00 CHECK (balance >= 0),
    account_type ENUM('Savings', 'Checking', 'Business') NOT NULL,
    opened_date DATE DEFAULT (CURRENT_DATE),
    last_activity TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- Stored procedures
DELIMITER //

CREATE PROCEDURE GetStudentCount()
BEGIN
    SELECT COUNT(*) AS total_students FROM students;
END //

CREATE PROCEDURE GetStudentsByCity(IN city_name VARCHAR(50))
BEGIN
    SELECT name, age, email 
    FROM students 
    WHERE city = city_name;
END //

CREATE PROCEDURE AddStudent(
    IN student_name VARCHAR(100),
    IN student_age INT,
    IN student_email VARCHAR(100),
    IN student_city VARCHAR(50)
)
BEGIN
    INSERT INTO students (name, age, email, city)
    VALUES (student_name, student_age, student_email, student_city);
    
    SELECT LAST_INSERT_ID() AS new_student_id;
END //

DELIMITER ;

-- Calling procedures
CALL GetStudentCount();
CALL GetStudentsByCity('London');
CALL AddStudent('David Wilson', 23, 'david@email.com', 'Berlin');

-- Functions
DELIMITER //

CREATE FUNCTION GetAverageAge() 
RETURNS DECIMAL(5,2)
READS SQL DATA
DETERMINISTIC
BEGIN
    DECLARE avg_age DECIMAL(5,2);
    SELECT AVG(age) INTO avg_age FROM students;
    RETURN avg_age;
END //

CREATE FUNCTION IsAdult(age INT) 
RETURNS BOOLEAN
DETERMINISTIC
BEGIN
    RETURN age >= 18;
END //

DELIMITER ;

-- Using functions
SELECT name, age, IsAdult(age) AS is_adult FROM students;
SELECT GetAverageAge() AS average_age;

-- Triggers
DELIMITER //

-- Before insert trigger
CREATE TRIGGER before_student_insert
BEFORE INSERT ON students
FOR EACH ROW
BEGIN
    IF NEW.age < 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Age cannot be negative';
    END IF;
    
    SET NEW.name = TRIM(NEW.name);
    SET NEW.email = LOWER(NEW.email);
END //

-- After insert trigger
CREATE TRIGGER after_student_insert
AFTER INSERT ON students
FOR EACH ROW
BEGIN
    INSERT INTO audit_log (table_name, action, record_id, changed_at)
    VALUES ('students', 'INSERT', NEW.id, NOW());
END //

-- Before update trigger
CREATE TRIGGER before_student_update
BEFORE UPDATE ON students
FOR EACH ROW
BEGIN
    IF NEW.age < OLD.age THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Age cannot decrease';
    END IF;
END //

DELIMITER ;

DROP FUNCTION IF EXISTS GetAverageAge;
DROP TRIGGER IF EXISTS before_student_insert;

Transactions & Backup

Data consistency and protection

-- Basic transaction
START TRANSACTION;

INSERT INTO students (name, age, email) 
VALUES ('Emma Brown', 24, 'emma@email.com');

INSERT INTO enrollments (student_id, course_id) 
VALUES (LAST_INSERT_ID(), 1);

COMMIT;

-- Transaction with error handling
START TRANSACTION;

BEGIN
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        RESIGNAL;
    END;
    
    UPDATE accounts SET balance = balance - 100 WHERE id = 1;
    UPDATE accounts SET balance = balance + 100 WHERE id = 2;
    
    COMMIT;
END;

-- Savepoints
START TRANSACTION;

INSERT INTO students (name, age) VALUES ('Frank Miller', 21);
SAVEPOINT student_added;

INSERT INTO enrollments (student_id, course_id) VALUES (LAST_INSERT_ID(), 1);

-- If enrollment fails, rollback to savepoint
-- ROLLBACK TO student_added;

COMMIT;

-- Transaction isolation levels
SET TRANSACTION ISOLATION LEVEL READ COMMITTED;
START TRANSACTION;
SELECT * FROM students;

END //

import mysql.connector
from mysql.connector import Error

try:
    # Connect to MySQL
    connection = mysql.connector.connect(
        host='localhost',
        user='root',
        password='your_password',
        database='school'
    )
    
    if connection.is_connected():
        cursor = connection.cursor()
        
        # Execute query
        cursor.execute("SELECT * FROM students")
        students = cursor.fetchall()
        
        for student in students:
            print(student)
            
        # Insert data
        insert_query = """
        INSERT INTO students (name, age, email, city)
        VALUES (%s, %s, %s, %s)
        """
        student_data = ('John Doe', 22, 'john@email.com', 'New York')
        cursor.execute(insert_query, student_data)
        connection.commit()
        
except Error as e:
    print(f"Error: {e}")
finally:
    if connection.is_connected():
        cursor.close()
        connection.close()

import mysql from 'mysql2/promise';

async function connectMySQL() {
    try {
        // Create connection
        const connection = await mysql.createConnection({
            host: 'localhost',
            user: 'root',
            password: 'your_password',
            database: 'school'
        });

        // Execute query
        const [students] = await connection.execute(
            'SELECT * FROM students WHERE age > ?',
            [20]
        );

        console.log(students);

        // Insert data
        const [result] = await connection.execute(
            `INSERT INTO students (name, age, email, city)
             VALUES (?, ?, ?, ?)`,
            ['Jane Smith', 25, 'jane@email.com', 'London']
        );

        console.log('Inserted student ID:', result.insertId);

        await connection.end();
    } catch (error) {
        console.error('Database error:', error);
    }
}
