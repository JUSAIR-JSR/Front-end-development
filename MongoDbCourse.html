Query Operators

Powerful operators for filtering data

// Comparison Operators
db.students.find({ age: { $eq: 20 } });      // Equal to
db.students.find({ age: { $ne: 20 } });      // Not equal to
db.students.find({ age: { $gt: 20 } });      // Greater than
db.students.find({ age: { $gte: 20 } });     // Greater than or equal
db.students.find({ age: { $lt: 25 } });      // Less than
db.students.find({ age: { $lte: 25 } });     // Less than or equal
db.students.find({ age: { $in: [18, 20, 22] } });  // In array
db.students.find({ age: { $nin: [18, 19] } });     // Not in array

// Logical Operators
// AND (implicit)
db.students.find({ age: { $gt: 18 }, enrolled: true });

// AND (explicit)
db.students.find({
  $and: [
    { age: { $gt: 18 } },
    { enrolled: true }
  ]
});

// OR
db.students.find({
  $or: [
    { age: { $lt: 20 } },
    { age: { $gt: 25 } }
  ]
});

// NOR (neither condition true)
db.students.find({
  $nor: [
    { age: { $lt: 20 } },
    { enrolled: false }
  ]
});

// NOT
db.students.find({ age: { $not: { $gt: 20 } } });

// Element Operators
db.students.find({ email: { $exists: true } });      // Field exists
db.students.find({ age: { $type: "number" } });      // Field type
db.students.find({ name: { $regex: /^J/ } });        // Regular expression
db.students.find({ name: { $regex: "John", $options: "i" } }); // Case insensitive

// Array Operators
db.students.find({ courses: { $size: 2 } });         // Array size
db.students.find({ courses: { $all: ["Math", "Science"] } }); // All elements
db.students.find({ courses: "Math" });               // Contains element
db.students.find({ "courses.0": "Math" });           // Element at position

// Evaluation Operators
db.students.find({ $expr: { $gt: ["$age", "$minAge"] } }); // Expression
db.students.find({ $jsonSchema: { ... } });          // JSON schema validation
db.students.find({ name: { $mod: [5, 0] } });        // Modulo operation

Aggregation Pipeline

Powerful data processing and transformation

// Basic aggregation example
db.students.aggregate([
  // Stage 1: Filter documents
  {
    $match: {
      enrolled: true,
      age: { $gte: 18 }
    }
  },
  // Stage 2: Group and calculate
  {
    $group: {
      _id: "$city",
      totalStudents: { $sum: 1 },
      averageAge: { $avg: "$age" },
      maxAge: { $max: "$age" },
      minAge: { $min: "$age" },
      names: { $push: "$name" }
    }
  },
  // Stage 3: Sort results
  {
    $sort: { totalStudents: -1 }
  }
]);

// Complex aggregation with multiple stages
db.orders.aggregate([
  // Match specific orders
  {
    $match: {
      status: "completed",
      orderDate: { $gte: new Date("2024-01-01") }
    }
  },
  // Unwind array (orders with multiple items)
  {
    $unwind: "$items"
  },
  // Lookup product details
  {
    $lookup: {
      from: "products",
      localField: "items.productId",
      foreignField: "_id",
      as: "productDetails"
    }
  },
  // Unwind product details
  {
    $unwind: "$productDetails"
  },
  // Calculate total for each item
  {
    $addFields: {
      "items.totalPrice": {
        $multiply: ["$items.quantity", "$productDetails.price"]
      }
    }
  },
  // Group back by order
  {
    $group: {
      _id: "$_id",
      customerId: { $first: "$customerId" },
      orderDate: { $first: "$orderDate" },
      totalAmount: { $sum: "$items.totalPrice" },
      items: { $push: "$items" }
    }
  },
  // Lookup customer details
  {
    $lookup: {
      from: "customers",
      localField: "customerId",
      foreignField: "_id",
      as: "customer"
    }
  },
  // Project final output
  {
    $project: {
      _id: 1,
      orderDate: 1,
      totalAmount: 1,
      "customer.name": 1,
      "customer.email": 1,
      items: 1
    }
  },
  // Sort by total amount
  {
    $sort: { totalAmount: -1 }
  },
  // Limit results
  {
    $limit: 10
  }
]);

// Common aggregation stages:
// $match - Filter documents (like WHERE)
// $group - Group by field and calculate
// $sort - Sort documents
// $project - Select/rename fields (like SELECT)
// $limit - Limit number of documents
// $skip - Skip documents
// $unwind - Deconstruct array
// $lookup - Join collections
// $addFields - Add new fields
// $count - Count documents
// $facet - Multiple pipelines
// $bucket - Group into buckets
// $graphLookup - Recursive search

Indexes & Performance

Optimize query performance with indexes

// Single field index
db.students.createIndex({ email: 1 });        // Ascending
db.students.createIndex({ age: -1 });         // Descending

// Compound index
db.students.createIndex({ city: 1, age: -1 });

// Unique index
db.students.createIndex({ email: 1 }, { unique: true });

// Partial index (only on subset)
db.students.createIndex(
  { email: 1 },
  { partialFilterExpression: { enrolled: true } }
);

// TTL index (auto-delete after time)
db.sessions.createIndex(
  { createdAt: 1 },
  { expireAfterSeconds: 3600 } // 1 hour
);

// Text index (for text search)
db.articles.createIndex(
  { title: "text", content: "text" }
);

// Use text search
db.articles.find({
  $text: { $search: "mongodb tutorial" }
});

// Geospatial index
db.places.createIndex({ location: "2dsphere" });

// Find nearby places
db.places.find({
  location: {
    $near: {
      $geometry: {
        type: "Point",
        coordinates: [longitude, latitude]
      },
      $maxDistance: 1000 // meters
    }
  }
});

// View indexes
db.students.getIndexes();

// Check query performance
db.students.find({ email: "test@example.com" }).explain("executionStats");

// Drop index
db.students.dropIndex("email_1");

// Index performance tips:
// 1. Create indexes on frequently queried fields
// 2. Use compound indexes for common query patterns
// 3. Monitor index usage with $indexStats
// 4. Consider partial indexes for large collections
// 5. Use covered queries (projection only indexed fields)

// Check index usage statistics
db.students.aggregate([{ $indexStats: {} }]);

// Force index usage
db.students.find({ email: "test@example.com" }).hint({ email: 1 });

Data Modeling

Embedded vs Referenced documents

// EMBEDDED DOCUMENTS (Denormalization)
// Good for: One-to-few, data that's accessed together
db.users.insertOne({
  _id: 1,
  name: "John Doe",
  address: {
    street: "123 Main St",
    city: "New York",
    country: "USA",
    zipCode: "10001"
  },
  contacts: [
    {
      type: "email",
      value: "john@example.com"
    },
    {
      type: "phone", 
      value: "+1234567890"
    }
  ]
});

// Advantages:
// - Single query gets all data
// - Atomic updates
// - Better read performance

// Disadvantages:
// - Document size limits (16MB)
// - Hard to query embedded fields separately

// REFERENCED DOCUMENTS (Normalization)
// Good for: One-to-many, many-to-many, large datasets

// Users collection
db.users.insertOne({
  _id: 1,
  name: "John Doe",
  email: "john@example.com"
});

// Posts collection  
db.posts.insertMany([
  {
    _id: 1,
    title: "First Post",
    content: "Hello World!",
    authorId: 1,  // Reference to user
    tags: ["mongodb", "tutorial"],
    createdAt: new Date()
  },
  {
    _id: 2, 
    title: "Second Post",
    content: "MongoDB is awesome!",
    authorId: 1,
    tags: ["database", "nosql"],
    createdAt: new Date()
  }
]);

// Advantages:
// - Smaller documents
// - Flexible queries
// - No document size issues

// Disadvantages:
// - Multiple queries needed ($lookup)
// - More complex data retrieval

// HYBRID APPROACH
// Embed frequently accessed data, reference less used data
db.products.insertOne({
  _id: 1,
  name: "Laptop",
  price: 999.99,
  // Embedded: Frequently accessed details
  specifications: {
    brand: "Dell",
    processor: "Intel i7",
    ram: "16GB",
    storage: "512GB SSD"
  },
  // Reference: Less frequently accessed
  supplierId: 123,
  categoryIds: [1, 2]
});

// Query with $lookup (like SQL JOIN)
db.posts.aggregate([
  {
    $lookup: {
      from: "users",           // Collection to join
      localField: "authorId",  // Field from posts
      foreignField: "_id",     // Field from users  
      as: "author"             // Output array field
    }
  },
  {
    $unwind: "$author"         // Convert array to object
  }
]);

// Many-to-Many relationships
db.students.insertOne({
  _id: 1,
  name: "Alice",
  courseIds: [101, 102, 103]  // Array of references
});

db.courses.insertMany([
  { _id: 101, name: "Math", instructor: "Dr. Smith" },
  { _id: 102, name: "Science", instructor: "Dr. Johnson" },
  { _id: 103, name: "History", instructor: "Dr. Brown" }
]);

// Query many-to-many with $lookup
db.students.aggregate([
  {
    $lookup: {
      from: "courses",
      localField: "courseIds", 
      foreignField: "_id",
      as: "courses"
    }
  }
]);

Advanced Features

Transactions, change streams, and more

// TRANSACTIONS (ACID operations)
const session = db.getMongo().startSession();

session.startTransaction({
  readConcern: { level: "snapshot" },
  writeConcern: { w: "majority" }
});

try {
  const students = session.getDatabase("school").students;
  const accounts = session.getDatabase("school").accounts;
  
  // Update student
  students.updateOne(
    { _id: 1 },
    { $set: { status: "graduated" } },
    { session }
  );
  
  // Update account
  accounts.updateOne(
    { studentId: 1 },
    { $set: { balance: 0 } },
    { session }
  );
  
  // Commit if both operations succeed
  session.commitTransaction();
  console.log("Transaction committed successfully");
  
} catch (error) {
  // Rollback if any operation fails
  session.abortTransaction();
  console.log("Transaction aborted:", error);
  
} finally {
  session.endSession();
}

// CHANGE STREAMS (Real-time notifications)
const changeStream = db.students.watch();

changeStream.on("change", (change) => {
  console.log("Change detected:", change);
  
  switch (change.operationType) {
    case "insert":
      console.log("New student:", change.fullDocument);
      break;
    case "update":
      console.log("Updated student:", change.updateDescription);
      break;
    case "delete":
      console.log("Deleted student ID:", change.documentKey._id);
      break;
  }
});

// Filtered change stream
const filteredStream = db.students.watch([
  { $match: { "operationType": "insert" } }
]);

// BULK WRITE OPERATIONS
const bulkOps = [
  {
    insertOne: {
      document: {
        name: "Student A",
        age: 20,
        enrolled: true
      }
    }
  },
  {
    updateOne: {
      filter: { name: "Student B" },
      update: { $set: { age: 21 } }
    }
  },
  {
    deleteOne: {
      filter: { name: "Student C" }
    }
  }
];

db.students.bulkWrite(bulkOps);

// AGGREGATION ADD-ONS

// $facet - Multiple aggregations in one query
db.orders.aggregate([
  {
    $facet: {
      "totalSales": [
        { $group: { _id: null, total: { $sum: "$amount" } } }
      ],
      "topCustomers": [
        { $group: { _id: "$customerId", total: { $sum: "$amount" } } },
        { $sort: { total: -1 } },
        { $limit: 5 }
      ],
      "monthlySales": [
        {
          $group: {
            _id: { $month: "$orderDate" },
            total: { $sum: "$amount" }
          }
        }
      ]
    }
  }
]);

// $graphLookup - Recursive relationships
db.employees.aggregate([
  {
    $match: { name: "CEO" }
  },
  {
    $graphLookup: {
      from: "employees",
      startWith: "$_id",
      connectFromField: "_id",
      connectToField: "managerId",
      as: "team",
      maxDepth: 3,
      depthField: "level"
    }
  }
]);

// TEXT SEARCH with scoring
db.articles.aggregate([
  {
    $match: {
      $text: { $search: "mongodb performance" }
    }
  },
  {
    $addFields: {
      score: { $meta: "textScore" }
    }
  },
  {
    $sort: { score: -1 }
  }
]);

Mongoose ODM Basics

Object Data Modeling for MongoDB with Node.js

// Installation: npm install mongoose

// Basic connection
const mongoose = require('mongoose');

// Connect to MongoDB
mongoose.connect('mongodb://localhost:27017/school', {
  useNewUrlParser: true,
  useUnifiedTopology: true
});

// Connection events
mongoose.connection.on('connected', () => {
  console.log('Mongoose connected to MongoDB');
});

mongoose.connection.on('error', (err) => {
  console.log('Mongoose connection error:', err);
});

mongoose.connection.on('disconnected', () => {
  console.log('Mongoose disconnected');
});

// Define a Schema
const studentSchema = new mongoose.Schema({
  name: {
    type: String,
    required: [true, 'Name is required'],
    trim: true,
    minlength: [2, 'Name must be at least 2 characters'],
    maxlength: [50, 'Name cannot exceed 50 characters']
  },
  age: {
    type: Number,
    required: true,
    min: [5, 'Age must be at least 5'],
    max: [120, 'Age cannot exceed 120']
  },
  email: {
    type: String,
    required: true,
    unique: true,
    lowercase: true,
    match: [/^\w+([.-]?\w+)*@\w+([.-]?\w+)*(\.\w{2,3})+$/, 'Please enter a valid email']
  },
  courses: [{
    type: String,
    enum: ['Math', 'Science', 'History', 'Art', 'Physics']
  }],
  address: {
    street: String,
    city: String,
    country: {
      type: String,
      default: 'USA'
    },
    zipCode: String
  },
  enrolled: {
    type: Boolean,
    default: false
  },
  gpa: {
    type: Number,
    min: 0,
    max: 4.0,
    default: 0
  }
}, {
  timestamps: true // Adds createdAt and updatedAt
});

// Create Model
const Student = mongoose.model('Student', studentSchema);

// Custom instance methods
studentSchema.methods.getInfo = function() {
  return `${this.name} (${this.age}) - ${this.email}`;
};

studentSchema.methods.isAdult = function() {
  return this.age >= 18;
};

// Static methods
studentSchema.statics.findByEmail = function(email) {
  return this.find({ email: new RegExp(email, 'i') });
};

studentSchema.statics.findAdults = function() {
  return this.find({ age: { $gte: 18 } });
};

// Virtuals (computed properties)
studentSchema.virtual('fullAddress').get(function() {
  return `${this.address.street}, ${this.address.city}, ${this.address.country}`;
});

// Middleware (Hooks)
studentSchema.pre('save', function(next) {
  console.log('About to save student:', this.name);
  next();
});

studentSchema.post('save', function(doc) {
  console.log('Student saved:', doc.name);
});

// Query middleware
studentSchema.pre('find', function(next) {
  this.start = Date.now();
  next();
});

studentSchema.post('find', function(docs) {
  console.log(`Query took ${Date.now() - this.start}ms`);
});

Express.js API with MongoDB

Build RESTful APIs with Express and Mongoose

// server.js
const express = require('express');
const mongoose = require('mongoose');
const app = express();

// Middleware
app.use(express.json());

// Connect to MongoDB
mongoose.connect('mongodb://localhost:27017/school-api', {
  useNewUrlParser: true,
  useUnifiedTopology: true
});

// Student Model (in models/Student.js)
const studentSchema = new mongoose.Schema({
  name: { type: String, required: true },
  age: { type: Number, required: true },
  email: { type: String, required: true, unique: true },
  courses: [String],
  enrolled: { type: Boolean, default: false }
}, { timestamps: true });

const Student = mongoose.model('Student', studentSchema);

// Routes

// GET /students - Get all students
app.get('/students', async (req, res) => {
  try {
    const { page = 1, limit = 10, search, enrolled } = req.query;
    
    const query = {};
    if (search) {
      query.name = { $regex: search, $options: 'i' };
    }
    if (enrolled !== undefined) {
      query.enrolled = enrolled === 'true';
    }
    
    const students = await Student.find(query)
      .limit(limit * 1)
      .skip((page - 1) * limit)
      .sort({ createdAt: -1 });
