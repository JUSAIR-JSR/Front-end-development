Functions & Typing

Type-safe function definitions and parameters

// Function with typed parameters and return
function add(a: number, b: number): number {
  return a + b;
}

// Arrow function
const multiply = (x: number, y: number): number => x * y;

// Optional parameters
function greet(name: string, greeting?: string): string {
  return `${greeting || 'Hello'}, ${name}!`;
}

// Default parameters
function createUser(name: string, role: string = "user"): object {
  return { name, role };
}

// Rest parameters
function join(separator: string, ...parts: string[]): string {
  return parts.join(separator);
}

// Function type
type MathOperation = (a: number, b: number) => number;
let operation: MathOperation = add;

// Function overloading
function processInput(input: string): string;
function processInput(input: number): number;
function processInput(input: string | number): string | number {
  return typeof input === 'string' ? input.toUpperCase() : input * 2;
}

Interfaces & Advanced Types

Interfaces & Type Aliases

Define object shapes and complex types

// Interface for object shape
interface User {
  id: number;
  name: string;
  email: string;
  age?: number; // Optional property
  readonly createdAt: Date; // Readonly property
}

// Function using interface
function createUser(user: User): void {
  console.log(`Creating user: ${user.name}`);
}

// Extending interfaces
interface Admin extends User {
  permissions: string[];
  isActive: boolean;
}

// Type alias for unions
type ID = string | number;
type Point = { x: number; y: number };

// Interface vs Type - prefer interface for objects
interface Product {
  id: ID;
  name: string;
  price: number;
}

// Index signatures
interface StringDictionary {
  [key: string]: string;
}

const dict: StringDictionary = {
  name: "John",
  email: "john@example.com"
};

Unions, Intersections & Generics

Advanced type composition and generic programming

// Union types
type Status = "pending" | "success" | "error";
type ID = string | number;

function handleStatus(status: Status): void {
  if (status === "success") {
    console.log("Operation completed successfully!");
  }
}

// Type narrowing
function processInput(input: string | number): void {
  if (typeof input === "string") {
    console.log(input.toUpperCase());
  } else {
    console.log(input.toFixed(2));
  }
}

// Intersection types
type A = { a: number };
type B = { b: string };
type C = A & B; // { a: number; b: string }

// Generics
function identity(arg: T): T {
  return arg;
}

const result = identity("Hello Generics!");

// Generic interfaces
interface ApiResponse {
  data: T;
  status: number;
  message: string;
}

// Generic constraints
interface HasLength {
  length: number;
}

function getLength(item: T): number {
  return item.length;
}

// Generic classes
class DataStore {
  private data: T[] = [];

  add(item: T): void {
    this.data.push(item);
  }

  getAll(): T[] {
    return this.data;
  }
}

// Custom hook with types
interface UseCounterReturn {
  count: number;
  increment: () => void;
  decrement: () => void;
  reset: () => void;
}

function useCounter(initialValue: number = 0): UseCounterReturn {
  const [count, setCount] = useState(initialValue);

  const increment = () => setCount(prev => prev + 1);
  const decrement = () => setCount(prev => prev - 1);
  const reset = () => setCount(initialValue);

  return { count, increment, decrement, reset };
}

TypeScript with Node.js & Express

Express Server with TypeScript

import express, { Request, Response, NextFunction } from 'express';
import cors from 'cors';

const app = express();

// Middleware
app.use(cors());
app.use(express.json());

// Custom middleware with types
interface AuthRequest extends Request {
  user?: { id: number; name: string };
}

const authMiddleware = (req: AuthRequest, res: Response, next: NextFunction) => {
  const token = req.headers.authorization;
  
  if (token) {
    req.user = { id: 1, name: 'John Doe' };
  }
  
  next();
};

// DTOs (Data Transfer Objects)
interface CreateUserDto {
  name: string;
  email: string;
  password: string;
}

interface UpdateUserDto {
  name?: string;
  email?: string;
}

// Routes with typed request bodies
app.post('/api/users', (req: Request<{}, {}, CreateUserDto>, res: Response) => {
  const { name, email, password } = req.body;
  
  // TypeScript knows the shape of req.body
  const newUser = {
    id: Date.now(),
    name,
    email,
    createdAt: new Date()
  };
  
  res.status(201).json(newUser);
});

app.put('/api/users/:id', (req: Request<{ id: string }, {}, UpdateUserDto>, res: Response) => {
  const userId = req.params.id;
  const updates = req.body;
  
  res.json({ 
    message: `User ${userId} updated`,
    updates 
  });
});

// Error handling middleware
app.use((err: Error, req: Request, res: Response, next: NextFunction) => {
  console.error(err.stack);
  res.status(500).json({ error: 'Something went wrong!' });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
});

Utility Types & Patterns

// Utility Types Examples
interface User {
  id: number;
  name: string;
  email: string;
  age: number;
  createdAt: Date;
}

// Partial - all properties optional
type UserUpdate = Partial;

// Required - all properties required
type CompleteUser = Required;

// Pick - select specific properties
type UserPreview = Pick;

// Omit - remove specific properties
type UserWithoutId = Omit;

// Record - map keys to type
type UserMap = Record;

// ReturnType - get function return type
type ApiResponse = ReturnType;

// Conditional Types
type IsString = T extends string ? true : false;
type A = IsString<'hello'>; // true
type B = IsString<123>; // false

// Mapped Types
type ReadonlyUser = { readonly [K in keyof User]: User[K] };
type OptionalUser = { [K in keyof User]?: User[K] };

// Template Literal Types
type EventName = 'click' | 'hover' | 'scroll';
type HandlerName = `on${Capitalize}`;
// "onClick" | "onHover" | "onScroll"

// Real-world example: API client
interface ApiConfig {
  baseURL: string;
  timeout: number;
  headers: Record;
}

type ApiMethod = 'GET' | 'POST' | 'PUT' | 'DELETE';

async function apiRequest(
  method: ApiMethod,
  endpoint: string,
  data?: any
): Promise {
  const response = await fetch(endpoint, {
    method,
    body: data ? JSON.stringify(data) : undefined,
    headers: { 'Content-Type': 'application/json' }
  });
  
  if (!response.ok) {
    throw new Error(`API error: ${response.status}`);
  }
  
  return response.json() as Promise;
}

interface Todo {
  id: number;
  title: string;
  completed: boolean;
}

const todos = await apiRequest('GET', '/api/todos');
