Creating Database & Tables

Create your first database and tables

CREATE DATABASE students_db;

-- Connect to the database

-- Create a table
CREATE TABLE students (
  id SERIAL PRIMARY KEY,
  name VARCHAR(50) NOT NULL,
  age INT CHECK (age > 0),
  email VARCHAR(100) UNIQUE,

-- Create another table for courses

  id SERIAL PRIMARY KEY,

SQL Fundamentals

Constraints & Keys

Data integrity and relationships

-- Table with various constraints
CREATE TABLE users (
  id SERIAL PRIMARY KEY,                    -- Primary Key
  username VARCHAR(30) UNIQUE NOT NULL,     -- Unique & Not Null
  email VARCHAR(100) UNIQUE NOT NULL,
  age INT CHECK (age >= 0 AND age <= 150),  -- Check constraint
  status VARCHAR(20) DEFAULT 'active',      -- Default value
  created_at TIMESTAMP DEFAULT NOW()
);

-- Foreign Key example
CREATE TABLE orders (
  id SERIAL PRIMARY KEY,
  user_id INT NOT NULL REFERENCES users(id), -- Foreign Key
  amount DECIMAL(10,2) CHECK (amount > 0),
  order_date DATE DEFAULT CURRENT_DATE
);

-- Composite Primary Key
CREATE TABLE enrollments (
  student_id INT REFERENCES students(id),
  course_id INT REFERENCES courses(id),
  enrolled_date DATE DEFAULT CURRENT_DATE,
  grade CHAR(1) CHECK (grade IN ('A', 'B', 'C', 'D', 'F')),
  PRIMARY KEY (student_id, course_id)        -- Composite PK
);

🔐 Constraint Types

PRIMARY KEY - Unique identifier
FOREIGN KEY - References another table
UNIQUE - No duplicate values
NOT NULL - Value required
CHECK - Custom validation
DEFAULT - Automatic value

JOIN Operations

Combine data from multiple tables

-- Sample data setup
INSERT INTO enrollments (student_id, course_id, grade) VALUES
(1, 1, 'A'), (1, 2, 'B'), (2, 1, 'C'), (3, 3, 'A');

-- INNER JOIN (most common)
SELECT 
  s.name AS student_name,
  c.title AS course_title,
  e.grade
FROM enrollments e
INNER JOIN students s ON e.student_id = s.id
INNER JOIN courses c ON e.course_id = c.id;

-- LEFT JOIN (all students, even without enrollments)
SELECT 
  s.name AS student_name,
  c.title AS course_title
FROM students s
LEFT JOIN enrollments e ON s.id = e.student_id
LEFT JOIN courses c ON e.course_id = c.id;

-- RIGHT JOIN (all courses, even without students)
SELECT 
  c.title AS course_title,
  s.name AS student_name
FROM courses c
LEFT JOIN enrollments e ON c.id = e.course_id
LEFT JOIN students s ON e.student_id = s.id;

-- FULL OUTER JOIN (all records from both sides)
SELECT 
  s.name AS student_name,
  c.title AS course_title
FROM students s
FULL OUTER JOIN enrollments e ON s.id = e.student_id
FULL OUTER JOIN courses c ON e.course_id = c.id;

-- SELF JOIN (join table with itself)
CREATE TABLE employees (
  id SERIAL PRIMARY KEY,
  name VARCHAR(50),
  manager_id INT REFERENCES employees(id)
);

SELECT 
  e.name AS employee,
  m.name AS manager
FROM employees e
LEFT JOIN employees m ON e.manager_id = m.id;

Built-in Functions

Powerful functions for data manipulation

-- String Functions
SELECT 
  name,
  UPPER(name) AS upper_name,
  LOWER(name) AS lower_name,
  LENGTH(name) AS name_length,
  SUBSTRING(name FROM 1 FOR 3) AS first_three,
  REPLACE(name, ' ', '_') AS name_underscore
FROM students;

-- Date/Time Functions
SELECT 
  created_at,
  EXTRACT(YEAR FROM created_at) AS year,
  EXTRACT(MONTH FROM created_at) AS month,
  DATE_PART('dow', created_at) AS day_of_week,
  created_at + INTERVAL '1 day' AS tomorrow,
  AGE(created_at) AS age_interval
FROM students;

-- Mathematical Functions
SELECT 
  age,
  ROUND(age * 1.0, 2) AS age_decimal,
  ABS(age - 25) AS age_diff_from_25,
  POWER(age, 2) AS age_squared,
  SQRT(age) AS age_sqrt
FROM students;

-- Aggregate Functions
SELECT 
  COUNT(*) AS total_students,
  AVG(age) AS average_age,
  MIN(age) AS min_age,
  MAX(age) AS max_age,
  SUM(age) AS total_years,
  STDDEV(age) AS age_stddev
FROM students;

-- Conditional Functions
SELECT 
  name,
  age,
  CASE 
    WHEN age < 20 THEN 'Teen'
    WHEN age BETWEEN 20 AND 25 THEN 'Young Adult'
    ELSE 'Adult'
  END AS age_group,
  COALESCE(city, 'Not specified') AS city_name
FROM students;

Aggregations & Grouping

Summarize and analyze data

-- Basic GROUP BY
SELECT 
  city,
  COUNT(*) AS student_count,
  AVG(age) AS average_age
FROM students
GROUP BY city
ORDER BY student_count DESC;

-- GROUP BY with HAVING (filter groups)
SELECT 
  city,
  COUNT(*) AS student_count
FROM students
GROUP BY city
HAVING COUNT(*) > 1;

-- Multiple column grouping
SELECT 
  city,
  EXTRACT(YEAR FROM created_at) AS year,
  COUNT(*) AS count_per_year
FROM students
GROUP BY city, EXTRACT(YEAR FROM created_at)
ORDER BY city, year;

-- ROLLUP (subtotals)
SELECT 
  COALESCE(city, 'All Cities') AS city,
  COUNT(*) AS student_count
FROM students
GROUP BY ROLLUP (city)
ORDER BY city;

-- CUBE (all combinations)
SELECT 
  city,
  CASE 
    WHEN age < 22 THEN 'Under 22'
    ELSE '22 and over'
  END AS age_group,
  COUNT(*) AS count
FROM students
GROUP BY CUBE (city, age_group)
ORDER BY city, age_group;

-- Window Functions (advanced)
SELECT 
  name,
  age,
  city,
  RANK() OVER (PARTITION BY city ORDER BY age DESC) AS age_rank_in_city,
  AVG(age) OVER (PARTITION BY city) AS avg_age_in_city
FROM students;

Advanced PostgreSQL

Views

Virtual tables for simplified queries

-- Create a simple view
CREATE VIEW student_courses_view AS
SELECT 
  s.name AS student_name,
  s.email,
  c.title AS course_title,
  e.enrolled_date,
  e.grade
FROM students s
JOIN enrollments e ON s.id = e.student_id
JOIN courses c ON e.course_id = c.id;

-- Use the view
SELECT * FROM student_courses_view 
WHERE grade = 'A'
ORDER BY student_name;

-- Create a view with calculated columns
CREATE VIEW student_stats AS
SELECT 
  city,
  COUNT(*) AS total_students,
  ROUND(AVG(age), 2) AS average_age,
  MIN(age) AS youngest,
  MAX(age) AS oldest
FROM students
GROUP BY city;

-- Materialized View (stores results physically)
CREATE MATERIALIZED VIEW popular_courses AS
SELECT 
  c.title,
  COUNT(e.student_id) AS enrollment_count
FROM courses c
LEFT JOIN enrollments e ON c.id = e.course_id
GROUP BY c.id, c.title
ORDER BY enrollment_count DESC;

-- Refresh materialized view
REFRESH MATERIALIZED VIEW popular_courses;

-- Updateable view (with INSTEAD OF triggers)
CREATE VIEW active_students AS
SELECT id, name, email, city
FROM students
WHERE age BETWEEN 18 AND 25;

-- Drop view
DROP VIEW IF EXISTS student_courses_view;

Indexes

Improve query performance

-- Basic B-tree index (default)
CREATE INDEX idx_students_email ON students(email);
CREATE INDEX idx_students_city ON students(city);

-- Composite index
CREATE INDEX idx_students_city_age ON students(city, age);

-- Unique index
CREATE UNIQUE INDEX idx_unique_student_email ON students(email);

-- Partial index (only on subset of data)
CREATE INDEX idx_active_students ON students(name) 
WHERE age >= 18;

-- Expression index
CREATE INDEX idx_student_name_lower ON students(LOWER(name));

-- Hash index (good for equality comparisons)
CREATE INDEX idx_hash_city ON students USING HASH (city);

-- GIN index (for JSON/array data)
CREATE TABLE products (
  id SERIAL PRIMARY KEY,
  name VARCHAR(100),
  tags TEXT[],
  metadata JSONB
);

CREATE INDEX idx_gin_tags ON products USING GIN (tags);
CREATE INDEX idx_gin_metadata ON products USING GIN (metadata);

-- BRIN index (for large, sorted tables)
CREATE INDEX idx_brin_created ON students USING BRIN (created_at);

-- Check existing indexes
SELECT 
  tablename,
  indexname,
  indexdef
FROM pg_indexes
WHERE schemaname = 'public';

-- Drop index
DROP INDEX IF EXISTS idx_students_city;

⚡ Index Types

B-tree - Default, good for ranges
Hash - Fast equality lookups
GIN - JSON, arrays, full-text
GiST - Geometric data, full-text
BRIN - Large sorted tables
SP-GiST - Clustered data

Stored Procedures & Functions

Reusable database logic

-- SQL Function
CREATE OR REPLACE FUNCTION get_student_count()
RETURNS INTEGER AS $$
  SELECT COUNT(*) FROM students;
$$ LANGUAGE SQL;

-- Use the function
SELECT get_student_count();

-- Function with parameters
CREATE OR REPLACE FUNCTION get_students_by_city(city_name VARCHAR)
RETURNS TABLE(
  student_id INTEGER,
  student_name VARCHAR,
  student_age INTEGER
) AS $$
  SELECT id, name, age 
  FROM students 
  WHERE city = city_name;
$$ LANGUAGE SQL;

-- Use parameterized function
SELECT * FROM get_students_by_city('London');

-- PL/pgSQL Function (procedural logic)
CREATE OR REPLACE FUNCTION calculate_age_statistics()
RETURNS TABLE(
  city_name VARCHAR,
  avg_age DECIMAL,
  min_age INTEGER,
  max_age INTEGER
) AS $$
BEGIN
  RETURN QUERY
  SELECT 
    s.city,
    ROUND(AVG(s.age), 2),
    MIN(s.age),
    MAX(s.age)
  FROM students s
  GROUP BY s.city;
END;
$$ LANGUAGE plpgsql;

-- Stored Procedure (can have transactions)
CREATE OR REPLACE PROCEDURE enroll_student(
  student_name VARCHAR,
  course_title VARCHAR
)
LANGUAGE plpgsql
AS $$
DECLARE
  s_id INTEGER;
  c_id INTEGER;
BEGIN
  -- Get student ID
  SELECT id INTO s_id FROM students WHERE name = student_name;
  
  -- Get course ID
  SELECT id INTO c_id FROM courses WHERE title = course_title;
  
  -- Insert enrollment
  INSERT INTO enrollments (student_id, course_id, enrolled_date)
  VALUES (s_id, c_id, CURRENT_DATE);
  
  COMMIT;
EXCEPTION
  WHEN others THEN
    ROLLBACK;
    RAISE NOTICE 'Error enrolling student: %', SQLERRM;
END;
$$;

-- Call procedure
CALL enroll_student('Alice Johnson', 'Database Fundamentals');

-- Function with conditional logic
CREATE OR REPLACE FUNCTION get_student_grade_level(age INTEGER)
RETURNS VARCHAR AS $$
BEGIN
  IF age < 18 THEN
    RETURN 'High School';
  ELSIF age BETWEEN 18 AND 22 THEN
    RETURN 'Undergraduate';
  ELSE
    RETURN 'Graduate';
  END IF;
END;
$$ LANGUAGE plpgsql;

Transactions

ACID properties and data consistency

-- Basic transaction
BEGIN;
  INSERT INTO students (name, age, email) 
  VALUES ('David Wilson', 23, 'david@email.com');
  
  INSERT INTO enrollments (student_id, course_id) 
  VALUES (currval('students_id_seq'), 1);
COMMIT;

-- Transaction with error handling
BEGIN;
  UPDATE accounts SET balance = balance - 100 WHERE id = 1;
  UPDATE accounts SET balance = balance + 100 WHERE id = 2;
  
  -- If anything fails, both updates are rolled back
COMMIT;

-- Transaction with savepoints
BEGIN;
  INSERT INTO students (name, age) VALUES ('Eva Brown', 21);
  SAVEPOINT student_added;
  
  INSERT INTO enrollments (student_id, course_id) 
  VALUES (currval('students_id_seq'), 1);
  
  -- If enrollment fails, rollback to savepoint
  -- ROLLBACK TO student_added;
  
COMMIT;

-- Transaction isolation levels
BEGIN TRANSACTION ISOLATION LEVEL READ COMMITTED;
  SELECT * FROM students WHERE city = 'London';
  -- Other transactions can see changes after commit
COMMIT;

BEGIN TRANSACTION ISOLATION LEVEL REPEATABLE READ;
  SELECT * FROM students;
  -- Guaranteed consistent reads within transaction
COMMIT;

BEGIN TRANSACTION ISOLATION LEVEL SERIALIZABLE;
  -- Highest isolation, prevents phantom reads
  SELECT COUNT(*) FROM students WHERE age > 20;
COMMIT;

-- Monitoring transactions
SELECT 
  pid,
  usename,
  application_name,
  state,
  query
FROM pg_stat_activity 
WHERE state = 'active';

-- Transaction in function
CREATE OR REPLACE FUNCTION transfer_funds(
  from_account INTEGER,
  to_account INTEGER,
  amount DECIMAL
) RETURNS BOOLEAN AS $$
BEGIN
  -- Start transaction
  BEGIN
    UPDATE accounts SET balance = balance - amount WHERE id = from_account;
    IF NOT FOUND THEN
      RAISE EXCEPTION 'Source account not found';
    END IF;
    
    UPDATE accounts SET balance = balance + amount WHERE id = to_account;
    IF NOT FOUND THEN
      RAISE EXCEPTION 'Destination account not found';
    END IF;
    
    -- Log the transaction
    INSERT INTO transactions (from_account, to_account, amount, timestamp)
    VALUES (from_account, to_account, amount, NOW());
    
    COMMIT;
    RETURN TRUE;
    
  EXCEPTION
    WHEN others THEN
      ROLLBACK;
      RETURN FALSE;
  END;
END;
$$ LANGUAGE plpgsql;

Student Management System

Complete database design with relationships

🗄️ Database Schema

-- Complete schema for student management system
CREATE TABLE departments (
  id SERIAL PRIMARY KEY,
  name VARCHAR(100) UNIQUE NOT NULL,
  head_of_department VARCHAR(100),
  budget DECIMAL(12,2) DEFAULT 0
);

CREATE TABLE instructors (
  id SERIAL PRIMARY KEY,
  name VARCHAR(100) NOT NULL,
  email VARCHAR(100) UNIQUE NOT NULL,
  department_id INTEGER REFERENCES departments(id),
  hire_date DATE DEFAULT CURRENT_DATE,
  salary DECIMAL(10,2) CHECK (salary > 0)
);

CREATE TABLE courses (
  id SERIAL PRIMARY KEY,
  title VARCHAR(200) NOT NULL,
  code VARCHAR(20) UNIQUE NOT NULL,
  credits INTEGER CHECK (credits BETWEEN 1 AND 5),
  department_id INTEGER REFERENCES departments(id),
  instructor_id INTEGER REFERENCES instructors(id),
  description TEXT,
  is_active BOOLEAN DEFAULT TRUE
);

CREATE TABLE students (
  id SERIAL PRIMARY KEY,
  student_id VARCHAR(20) UNIQUE NOT NULL,
  name VARCHAR(100) NOT NULL,
  email VARCHAR(100) UNIQUE NOT NULL,
  date_of_birth DATE,
  enrollment_date DATE DEFAULT CURRENT_DATE,
  major_department_id INTEGER REFERENCES departments(id),
  gpa DECIMAL(3,2) CHECK (gpa BETWEEN 0 AND 4.0)
);

CREATE TABLE enrollments (
  id SERIAL PRIMARY KEY,
  student_id INTEGER REFERENCES students(id),
  course_id INTEGER REFERENCES courses(id),
  enrollment_date DATE DEFAULT CURRENT_DATE,
  grade CHAR(2) CHECK (grade IN ('A', 'A-', 'B+', 'B', 'B-', 'C+', 'C', 'C-', 'D', 'F', 'W')),
  UNIQUE(student_id, course_id)
);

CREATE TABLE assignments (
  id SERIAL PRIMARY KEY,
  course_id INTEGER REFERENCES courses(id),
  title VARCHAR(200) NOT NULL,
  description TEXT,
  due_date TIMESTAMP,
  max_score INTEGER CHECK (max_score > 0)
);

CREATE TABLE grades (
  id SERIAL PRIMARY KEY,
  enrollment_id INTEGER REFERENCES enrollments(id),
  assignment_id INTEGER REFERENCES assignments(id),
  score DECIMAL(5,2) CHECK (score >= 0),
  submitted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  UNIQUE(enrollment_id, assignment_id)
);

🔗 Entity Relationships

Departments (1) ─── (∞) Courses
One department offers many courses
Instructors (1) ─── (∞) Courses
One instructor teaches many courses
Students (∞) ─── (∞) Courses
Many-to-many via enrollments
Courses (1) ─── (∞) Assignments
One course has many assignments

📊 Sample Queries

-- Students with their majors and GPAs
SELECT 
  s.name AS student_name,
  s.student_id,
  d.name AS major,
  s.gpa
FROM students s
JOIN departments d ON s.major_department_id = d.id
WHERE s.gpa > 3.5
ORDER BY s.gpa DESC;

-- Course enrollment counts
SELECT 
  c.title AS course_name,
  c.code,
  COUNT(e.student_id) AS enrollment_count
FROM courses c
LEFT JOIN enrollments e ON c.id = e.course_id
GROUP BY c.id, c.title, c.code
ORDER BY enrollment_count DESC;

-- Instructor workload
SELECT 
  i.name AS instructor_name,
  d.name AS department,
  COUNT(c.id) AS courses_taught,
  SUM(c.credits) AS total_credits
FROM instructors i
JOIN departments d ON i.department_id = d.id
LEFT JOIN courses c ON i.id = c.instructor_id
GROUP BY i.id, i.name, d.name
ORDER BY total_credits DESC;

Backup & Restore

Protect your data with proper backups

$ pg_dump students_db > backup.sql
# Backup entire database
$ pg_dump -t students students_db > students_backup.sql
# Backup specific table
$ pg_dump -Fc students_db > backup.custom
# Custom format (compressed)
$ pg_restore -d new_db backup.custom
# Restore from custom format
$ psql students_db < backup.sql
# Restore from SQL file
$ pg_dumpall > all_databases.sql
# Backup all databases

💾 Backup Strategies

Full backup - Complete database dump
Incremental - WAL archiving for point-in-time recovery
Logical - SQL statements (pg_dump)
Physical - File system level backup

User Management & Permissions

Control access and security

-- Create a new user
CREATE USER john_doe WITH PASSWORD 'securepassword123';

-- Create a role with specific privileges
CREATE ROLE read_only;
GRANT CONNECT ON DATABASE students_db TO read_only;
GRANT USAGE ON SCHEMA public TO read_only;
GRANT SELECT ON ALL TABLES IN SCHEMA public TO read_only;

-- Assign role to user
GRANT read_only TO john_doe;

-- Create admin role
CREATE ROLE db_admin;
GRANT ALL PRIVILEGES ON DATABASE students_db TO db_admin;
GRANT ALL PRIVILEGES ON ALL TABLES IN SCHEMA public TO db_admin;

-- Grant specific table permissions
GRANT SELECT, INSERT, UPDATE ON students TO john_doe;
GRANT DELETE ON students TO db_admin;

-- Revoke permissions
REVOKE DELETE ON students FROM john_doe;

-- Create group role
CREATE ROLE faculty;
GRANT faculty TO instructor1, instructor2;

-- Row Level Security (RLS)
ALTER TABLE students ENABLE ROW LEVEL SECURITY;

CREATE POLICY student_policy ON students
FOR ALL TO faculty
USING (major_department_id = current_setting('app.current_department_id')::INT);

-- View current privileges
SELECT 
  grantee,
  table_name,
  privilege_type
FROM information_schema.role_table_grants
WHERE table_name = 'students';

-- Drop user
DROP USER IF EXISTS john_doe;

  schemaname,
  tablename,
  n_dead_tup,
