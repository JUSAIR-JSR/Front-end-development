Loops

For loops, while loops, and loop control

# For loop with range
print("Counting from 1 to 5:")
for i in range(1, 6):
    print(i)

# For loop with list
fruits = ["apple", "banana", "cherry"]
print("\nFruits:")
for fruit in fruits:
    print(f"- {fruit}")

# For loop with enumerate
print("\nFruits with index:")
for index, fruit in enumerate(fruits):
    print(f"{index + 1}. {fruit}")

# While loop
print("\nCountdown:")
count = 5
while count > 0:
    print(count)
    count -= 1
print("Blast off!")

# Loop control: break and continue
print("\nNumbers 1-10 (skip 5):")
for i in range(1, 11):
    if i == 5:
        continue  # Skip 5
    if i == 8:
        break     # Stop at 8
    print(i)

# Nested loops
print("\nMultiplication table (1-3):")
for i in range(1, 4):
    for j in range(1, 4):
        print(f"{i} x {j} = {i * j}")

Functions

Defining and using functions in Python

# Basic function
def greet(name):
    return f"Hello, {name}!"

print(greet("Alice"))

# Function with default parameters
def create_user(name, age, city="Unknown"):
    return {
        "name": name,

    }

user2 = create_user("Sarah", 30, "London")

# Function with variable arguments
def calculate_sum(*numbers):
    return sum(numbers)

print("Sum of 1,2,3:", calculate_sum(1, 2, 3))
print("Sum of 1,2,3,4,5:", calculate_sum(1, 2, 3, 4, 5))

# Lambda functions (anonymous functions)
square = lambda x: x * x
double = lambda x: x * 2

print("Square of 5:", square(5))
print("Double of 10:", double(10))

# Using lambda with map and filter
numbers = [1, 2, 3, 4, 5]
squared = list(map(lambda x: x**2, numbers))
evens = list(filter(lambda x: x % 2 == 0, numbers))

print("Numbers:", numbers)
print("Squared:", squared)
print("Even numbers:", evens)

# Recursive function
def factorial(n):
    if n == 0:
        return 1
    else:
        return n * factorial(n - 1)

print("Factorial of 5:", factorial(5))

Error Handling

Exception handling and debugging

# Basic try-except
try:
    x = 10 / 0
except ZeroDivisionError:
    print("Error: Cannot divide by zero!")

# Multiple exceptions
try:
    number = int("abc")
    result = 10 / number
except ValueError:
    print("Error: Invalid number format!")
except ZeroDivisionError:
    print("Error: Division by zero!")
except Exception as e:
    print(f"Unexpected error: {e}")

# Try-except-else-finally
def divide_numbers(a, b):
    try:
        result = a / b
    except ZeroDivisionError:
        print("Cannot divide by zero!")
        return None
    except TypeError:
        print("Both arguments must be numbers!")
        return None
    else:
        print(f"Division successful: {a} / {b} = {result}")
        return result
    finally:
        print("This always executes")

# Test the function
print("Test 1:", divide_numbers(10, 2))
print("Test 2:", divide_numbers(10, 0))
print("Test 3:", divide_numbers(10, "2"))

# Raising exceptions
def check_age(age):
    if age < 0:
        raise ValueError("Age cannot be negative")
    if age < 18:
        raise ValueError("Must be 18 or older")
    return "Access granted"

try:
    print(check_age(25))
    print(check_age(16))
except ValueError as e:
    print(f"Error: {e}")

# Custom exception
class NegativeNumberError(Exception):
    pass

def check_positive(number):
    if number < 0:
        raise NegativeNumberError("Number must be positive")
    return number

try:
    print(check_positive(5))
    print(check_positive(-3))
except NegativeNumberError as e:
    print(f"Custom error: {e}")

Object-Oriented Programming

Classes & Objects

Creating classes, objects, and methods

# Basic class
class Person:
    # Constructor
    def __init__(self, name, age):
        self.name = name
        self.age = age
    
    # Instance method
    def introduce(self):
        return f"Hi, I'm {self.name} and I'm {self.age} years old"
    
    # Another method
    def have_birthday(self):
        self.age += 1
        return f"Happy birthday! Now I'm {self.age}"

# Create objects
person1 = Person("Alice", 25)
person2 = Person("Bob", 30)

print(person1.introduce())
print(person2.introduce())
print(person1.have_birthday())

# Class with class attribute
class Car:
    # Class attribute (shared by all instances)
    wheels = 4
    
    def __init__(self, brand, model, year):
        # Instance attributes
        self.brand = brand
        self.model = model
        self.year = year
        self.speed = 0
    
    def accelerate(self, amount):
        self.speed += amount
        return f"Accelerating to {self.speed} km/h"
    
    def brake(self, amount):
        self.speed = max(0, self.speed - amount)
        return f"Slowing down to {self.speed} km/h"
    
    def info(self):
        return f"{self.year} {self.brand} {self.model}"

# Using the Car class
my_car = Car("Toyota", "Camry", 2022)
print(my_car.info())
print("Wheels:", my_car.wheels)
print(my_car.accelerate(50))
print(my_car.brake(20))

# Inheritance
class ElectricCar(Car):
    def __init__(self, brand, model, year, battery_size):
        super().__init__(brand, model, year)
        self.battery_size = battery_size
        self.charge_level = 100
    
    def charge(self, amount):
        self.charge_level = min(100, self.charge_level + amount)
        return f"Charged to {self.charge_level}%"
    
    def info(self):  # Method overriding
        return f"{super().info()} (Electric)"

# Using ElectricCar
tesla = ElectricCar("Tesla", "Model S", 2023, 100)
print(tesla.info())
print(tesla.charge(50))

File Handling

Reading and writing files in Python

# Writing to a file
print("Writing to file...")
with open("example.txt", "w") as file:
    file.write("Hello, Python!\n")
    file.write("This is a sample file.\n")
    file.write("Python file handling is easy!\n")

print("File written successfully!")

# Reading from a file
print("\nReading entire file:")
with open("example.txt", "r") as file:
    content = file.read()
    print(content)

print("\nReading line by line:")
with open("example.txt", "r") as file:
    for line_num, line in enumerate(file, 1):
        print(f"Line {line_num}: {line.strip()}")

# Appending to a file
print("\nAppending to file...")
with open("example.txt", "a") as file:
    file.write("This line was appended!\n")

print("Content after appending:")
with open("example.txt", "r") as file:
    print(file.read())

# Working with CSV files
import csv

# Writing CSV
print("\nCreating CSV file...")
with open("data.csv", "w", newline='') as file:
    writer = csv.writer(file)
    writer.writerow(["Name", "Age", "City"])
    writer.writerow(["Alice", 25, "New York"])
    writer.writerow(["Bob", 30, "London"])
    writer.writerow(["Charlie", 35, "Tokyo"])

# Reading CSV
print("Reading CSV file:")
with open("data.csv", "r") as file:
    reader = csv.reader(file)
    for row in reader:
        print(row)

# Working with JSON
import json

# Writing JSON
data = {
    "name": "John Doe",
    "age": 30,
    "hobbies": ["reading", "coding", "gaming"],
    "address": {
        "street": "123 Main St",
        "city": "Boston"
    }
}

print("\nWriting JSON file...")
with open("data.json", "w") as file:
    json.dump(data, file, indent=2)

# Reading JSON
print("Reading JSON file:")
with open("data.json", "r") as file:
    loaded_data = json.load(file)
    print(json.dumps(loaded_data, indent=2))

Flask Web Framework

Building web applications with Flask

# app.py
from flask import Flask, jsonify, request, render_template

app = Flask(__name__)

# Basic route
@app.route('/')
def home():
    return "Welcome to Flask!"

# JSON API route
@app.route('/api/users')
def get_users():
    users = [
        {"id": 1, "name": "Alice", "email": "alice@example.com"},
        {"id": 2, "name": "Bob", "email": "bob@example.com"}
    ]
    return jsonify(users)

# Route with parameters
@app.route('/api/users/')
def get_user(user_id):
    user = {"id": user_id, "name": "John Doe", "email": "john@example.com"}
    return jsonify(user)

# POST route
@app.route('/api/users', methods=['POST'])
def create_user():
    data = request.get_json()
    # Process the data (save to database, etc.)
    new_user = {
        "id": 3,
        "name": data.get('name'),
        "email": data.get('email')
    }
    return jsonify(new_user), 201

# Template rendering
@app.route('/hello/')
def hello(name):
    return render_template('hello.html', name=name)

if __name__ == '__main__':
    app.run(debug=True)
