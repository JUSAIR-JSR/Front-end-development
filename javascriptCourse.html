Variables & Data Types

Learn how to store and work with different types of data

Functions

Create reusable blocks of code with functions

// Regular Function
function greet(name) {
  return `Hello, ${name}!`;
}

// Arrow Function
const add = (a, b) => a + b;

// Function Expression
const multiply = function(a, b) {
  return a * b;
};

// Using functions
console.log(greet("JavaScript"));
console.log("Sum:", add(5, 3));
console.log("Product:", multiply(4, 7));

// Higher-order function
function operate(a, b, operation) {
  return operation(a, b);
}

console.log("Operation result:", operate(10, 5, add));

Arrays & Objects

Work with collections of data using arrays and objects

// Arrays
let numbers = [10, 20, 30, 40, 50];
let fruits = ["Apple", "Banana", "Orange"];

// Array Methods
numbers.push(60); // Add to end
numbers.pop(); // Remove from end
fruits.unshift("Mango"); // Add to start

console.log("Numbers:", numbers);
console.log("Fruits:", fruits);
console.log("First fruit:", fruits[0]);
console.log("Array length:", numbers.length);

// Objects
let user = {
  name: "Jusair",
  age: 21,
  city: "India",
  isStudent: true
};

console.log("User object:", user);
console.log("User name:", user.name);
console.log("User age:", user.age);

// Object methods
user.greet = function() {
  return `Hello, I'm ${this.name}`;
};

console.log(user.greet());

Control Flow

Control the flow of your program with conditions and loops

// If-else statements
let age = 20;

if (age >= 18) {
  console.log("You are an adult");
} else {
  console.log("You are a minor");
}

// Switch statement
let day = "Monday";

switch (day) {
  case "Monday":
    console.log("Start of the week");
    break;
  case "Friday":
    console.log("Weekend is near!");
    break;
  default:
    console.log("It's a regular day");
}

// For loop
console.log("For loop:");
for (let i = 1; i <= 5; i++) {
  console.log("Count:", i);
}

// While loop
console.log("While loop:");
let count = 1;
while (count <= 3) {
  console.log("While count:", count);
  count++;
}

// Array iteration
let numbers = [1, 2, 3, 4, 5];
console.log("Array iteration:");
numbers.forEach(num => console.log("Number:", num));

Event Handling

Respond to user interactions with event listeners

// Click event
document.getElementById('click-btn').addEventListener('click', function() {
  alert('Button clicked!');
  this.textContent = 'Clicked!';
  this.classList.add('bg-green-500');
});

// Input event
document.getElementById('text-input').addEventListener('input', function(e) {
  document.getElementById('input-value').textContent = e.target.value;
});

// Mouse events
const hoverArea = document.getElementById('hover-area');
hoverArea.addEventListener('mouseenter', () => {
  hoverArea.classList.add('bg-blue-500', 'text-white');
});
hoverArea.addEventListener('mouseleave', () => {
  hoverArea.classList.remove('bg-blue-500', 'text-white');
});

Async JavaScript & APIs

Work with asynchronous operations and fetch data from APIs

// Fetch API with async/await
async function fetchUserData() {
  try {
    const response = await fetch('https://jsonplaceholder.typicode.com/users/1');
    const user = await response.json();
    
    console.log('User data:', user);
    displayUserInfo(user);
  } catch (error) {
    console.error('Error fetching data:', error);
  }
}

function displayUserInfo(user) {
  const userInfo = document.getElementById('user-info');
  userInfo.innerHTML = `
    

      
${user.name}

      
Email: ${user.email}

      
Phone: ${user.phone}

      
Website: ${user.website}

    

  `;
}

// Local Storage
localStorage.setItem('username', 'JavaScriptLearner');
const storedUser = localStorage.getItem('username');
console.log('Stored user:', storedUser);

ES6+ Modern Features

Explore modern JavaScript features and syntax

// Destructuring
const user = { name: "Jusair", age: 21, city: "India" };
const { name, age } = user;
console.log("Destructured:", name, age);

// Spread operator
const numbers = [1, 2, 3];
const newNumbers = [...numbers, 4, 5];
console.log("Spread array:", newNumbers);

// Template literals
const greeting = `Hello, my name is ${name} and I'm ${age} years old.`;
console.log(greeting);

// Arrow functions
const square = x => x * x;
console.log("Square of 5:", square(5));

// Classes
class Person {
  constructor(name, age) {
    this.name = name;
    this.age = age;
  }
  
  greet() {
    return `Hello, I'm ${this.name}`;
  }
}

const person1 = new Person("Alice", 25);
console.log(person1.greet());

// Modules (conceptual)
console.log("ES6 modules allow code splitting across files");
