my-next-app/
 ├─ pages/           # File-based routing

 │   │  └─ hello.js

 │   ├─ index.js     # Home page (/)

Data Fetching Methods

getStaticProps (SSG)

Static Generation - Build time data fetching

// pages/posts.js - Static page with posts
export default function Posts({ posts }) {
  return (
    <div>
      <h1>Blog Posts</h1>
      {posts.map(post => (
        <article key={post.id} className="mb-6">
          <h2>{post.title}</h2>
          <p>{post.body}</p>
        </article>
      ))}
    </div>
  );
}

// This function runs at build time
export async function getStaticProps() {
  // Fetch data from external API
  const res = await fetch('https://jsonplaceholder.typicode.com/posts');
  const posts = await res.json();

  // Return props that will be passed to the page component
  return {
    props: {
      posts: posts.slice(0, 10) // Only first 10 posts
    },
    // Re-generate page at most once every 60 seconds
    revalidate: 60 // ISR - Incremental Static Regeneration
  };
}

Server-Side Rendering - Data fetched on each request

// This function runs on every request
export async function getServerSideProps(context) {
  const { req, res, query, params } = context;
  
  // Access to request/response objects
  console.log('Request headers:', req.headers);
  
  // Fetch data that changes frequently
  const userResponse = await fetch('https://jsonplaceholder.typicode.com/users/1');
  const user = await userResponse.json();
  
  // Return props for the page
  return {
    props: {
      user,
      timestamp: new Date().toISOString()
    }
  };
}

getStaticPaths (Dynamic SSG)

Static Generation for dynamic routes

// pages/posts/[id].js - Dynamic post page
export default function Post({ post }) {
  if (!post) {
    return <div>Loading...</div>;
  }

  return (
    <article>
      <h1>{post.title}</h1>
      <p>{post.body}</p>
    </article>
  );
}

// Define which paths should be pre-rendered
export async function getStaticPaths() {
  // Fetch all posts to generate paths
  const res = await fetch('https://jsonplaceholder.typicode.com/posts');
  const posts = await res.json();

  // Generate paths for the first 20 posts
  const paths = posts.slice(0, 20).map(post => ({
    params: { id: post.id.toString() }
  }));

  return {
    paths,
    fallback: true // false: 404 for unknown paths, true: generate on demand
  };
}

// Fetch data for each pre-rendered path
export async function getStaticProps({ params }) {
  const res = await fetch(
    `https://jsonplaceholder.typicode.com/posts/${params.id}`
  );
  const post = await res.json();

  if (!post.id) {
    return { notFound: true }; // Show 404 page
  }

  return {
    props: { post },
    revalidate: 3600 // Re-generate every hour
  };
}

Client-Side Fetching (CSR)

Fetch data on the client side after page load

import { useState, useEffect } from 'react';

export default function ClientSideData() {
  const [data, setData] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    async function fetchData() {
      try {
        const response = await fetch('/api/posts');
        if (!response.ok) {
          throw new Error('Failed to fetch data');
        }
        const posts = await response.json();
        setData(posts);
      } catch (err) {
        setError(err.message);
      } finally {
        setLoading(false);
      }
    }

    fetchData();
  }, []);

  if (loading) return <div>Loading...</div>;
  if (error) return <div>Error: {error}</div>;

  return (
    <div>
      <h1>Client-Side Data</h1>
      {data.map(item => (
        <div key={item.id}>{item.title}</div>
      ))}
    </div>
  );
}

API Routes

Basic API Routes

Create backend endpoints within your Next.js app

// pages/api/hello.js - Simple API endpoint
export default function handler(req, res) {
  res.status(200).json({ 
    message: 'Hello from Next.js API!',
    timestamp: new Date().toISOString()
  });
}

// pages/api/users/index.js - Users API
export default function handler(req, res) {
  const { method } = req;

  switch (method) {
    case 'GET':
      // Get all users
      res.status(200).json([
        { id: 1, name: 'John Doe', email: 'john@example.com' },
        { id: 2, name: 'Jane Smith', email: 'jane@example.com' }
      ]);
      break;
    
    case 'POST':
      // Create new user
      const { name, email } = req.body;
      const newUser = {
        id: Date.now(),
        name,
        email,
        createdAt: new Date().toISOString()
      };
      res.status(201).json(newUser);
      break;
    
    default:
      res.setHeader('Allow', ['GET', 'POST']);
      res.status(405).end(`Method ${method} Not Allowed`);
  }
}

// Using the API from client
fetch('/api/users')
  .then(res => res.json())
  .then(users => console.log(users));

Dynamic API Routes

Handle dynamic parameters in API routes

// pages/api/users/[id].js - Single user API
export default function handler(req, res) {
  const { query, method } = req;
  const { id } = query;

  switch (method) {
    case 'GET':
      // Get user by ID
      const user = {
        id: parseInt(id),
        name: 'John Doe',
        email: 'john@example.com'
      };
      res.status(200).json(user);
      break;
    
    case 'PUT':
      // Update user
      const { name, email } = req.body;
      const updatedUser = {
        id: parseInt(id),
        name,
        email,
        updatedAt: new Date().toISOString()
      };
      res.status(200).json(updatedUser);
      break;
    
    case 'DELETE':
      // Delete user
      res.status(204).end();
      break;
    
    default:
      res.setHeader('Allow', ['GET', 'PUT', 'DELETE']);
      res.status(405).end(`Method ${method} Not Allowed`);
  }
}

// pages/api/posts/[...slug].js - Catch-all API route
export default function handler(req, res) {
  const { slug } = req.query;
  
  res.status(200).json({
    message: 'Catch-all API route',
    slug: slug,
    fullPath: slug.join('/')
  });
}
