Operators

Arithmetic, relational, logical, and assignment operators

#include <stdio.h>

int main() {
    int a = 10, b = 3;
    
    // Arithmetic operators
    printf("Arithmetic Operators:\n");
    printf("%d + %d = %d\n", a, b, a + b);
    printf("%d - %d = %d\n", a, b, a - b);
    printf("%d * %d = %d\n", a, b, a * b);
    printf("%d / %d = %d\n", a, b, a / b);
    printf("%d %% %d = %d\n", a, b, a % b);
    
    // Relational operators
    printf("\nRelational Operators:\n");
    printf("%d == %d: %d\n", a, b, a == b);
    printf("%d != %d: %d\n", a, b, a != b);
    printf("%d > %d: %d\n", a, b, a > b);
    printf("%d < %d: %d\n", a, b, a < b);
    printf("%d >= %d: %d\n", a, b, a >= b);
    printf("%d <= %d: %d\n", a, b, a <= b);
    
    // Logical operators
    int x = 1, y = 0;
    printf("\nLogical Operators:\n");
    printf("%d && %d: %d\n", x, y, x && y);
    printf("%d || %d: %d\n", x, y, x || y);
    printf("!%d: %d\n", x, !x);
    
    // Assignment operators
    int c = 5;
    printf("\nAssignment Operators:\n");
    c += 3;  printf("c += 3: %d\n", c);
    c -= 2;  printf("c -= 2: %d\n", c);
    c *= 2;  printf("c *= 2: %d\n", c);
    c /= 2;  printf("c /= 2: %d\n", c);
    
    // Increment/Decrement
    int count = 5;
    printf("\nIncrement/Decrement:\n");
    printf("count++: %d\n", count++);  // Post-increment
    printf("++count: %d\n", ++count);  // Pre-increment
    printf("count--: %d\n", count--);  // Post-decrement
    printf("--count: %d\n", --count);  // Pre-decrement
    
    // Bitwise operators
    unsigned int m = 5, n = 3;
    printf("\nBitwise Operators:\n");
    printf("%d & %d = %d\n", m, n, m & n);   // AND
    printf("%d | %d = %d\n", m, n, m | n);   // OR
    printf("%d ^ %d = %d\n", m, n, m ^ n);   // XOR
    printf("~%d = %d\n", m, ~m);             // NOT
    printf("%d << 1 = %d\n", m, m << 1);     // Left shift
    printf("%d >> 1 = %d\n", m, m >> 1);     // Right shift
    
    return 0;
}

Loops

For loops, while loops, and loop control

#include <stdio.h>

int main() {
    // For loop
    printf("For loop (1 to 5): ");
    for (int i = 1; i <= 5; i++) {
        printf("%d ", i);
    }
    printf("\n");
    
    // While loop
    printf("While loop (1 to 5): ");
    int j = 1;
    while (j <= 5) {
        printf("%d ", j);
        j++;
    }
    printf("\n");
    
    // Do-while loop
    printf("Do-while loop (1 to 5): ");
    int k = 1;
    do {
        printf("%d ", k);
        k++;
    } while (k <= 5);
    printf("\n");
    
    // Nested loops
    printf("\nMultiplication table (1-3):\n");
    for (int i = 1; i <= 3; i++) {
        for (int j = 1; j <= 3; j++) {
            printf("%d x %d = %d\n", i, j, i * j);
        }
    }
    
    // Loop control: break and continue
    printf("\nNumbers 1-10 (skip 5): ");
    for (int i = 1; i <= 10; i++) {
        if (i == 5) {
            continue;  // Skip 5
        }
        if (i == 8) {
            break;     // Stop at 8
        }
        printf("%d ", i);
    }
    printf("\n");
    
    // Infinite loop with break
    printf("Counting until 5: ");
    int counter = 1;
    while (1) {  // Infinite loop
        printf("%d ", counter);
        if (counter == 5) {
            break;
        }
        counter++;
    }
    printf("\n");
    
    // Array iteration with loops
    int numbers[] = {10, 20, 30, 40, 50};
    int size = sizeof(numbers) / sizeof(numbers[0]);
    
    printf("\nArray elements: ");
    for (int i = 0; i < size; i++) {
        printf("%d ", numbers[i]);
    }
    printf("\n");
    
    // Reverse array
    printf("Reverse array: ");
    for (int i = size - 1; i >= 0; i--) {
        printf("%d ", numbers[i]);
    }
    printf("\n");
    
    // Sum of array elements
    int sum = 0;
    for (int i = 0; i < size; i++) {
        sum += numbers[i];
    }
    printf("Sum of array: %d\n", sum);
    
    return 0;
}

Working with single and multi-dimensional arrays

#include <stdio.h>

int main() {
    // Single-dimensional arrays
    int numbers[5] = {10, 20, 30, 40, 50};
    float temperatures[3] = {36.5, 37.2, 35.8};

    int scores[5];  // Array of 5 integers
    scores[0] = 95;
    scores[1] = 87;
    scores[2] = 92;
    scores[3] = 78;

        for (int j = 0; j < 3; j++) {
            printf("%d ", matrix[i][j]);
        }
        printf("\n");

    char name[20] = "John Doe";
    char greeting[] = "Hello, World!";

    printf("Greeting: %s\n", greeting);
    printf("Name length: %zu\n", sizeof(name));
    printf("Greeting length: %zu\n", sizeof(greeting));

    name[0] = 'J';
    name[5] = 'D';
    printf("Modified name: %s\n", name);
    
    // Array of strings
    char fruits[3][20] = {
        "Apple",
        "Banana",
        "Orange"
    };
    
    printf("\nArray of Strings:\n");
    for (int i = 0; i < 3; i++) {
        printf("Fruit %d: %s\n", i + 1, fruits[i]);
    }
    
    // Array operations
    int arr[5] = {5, 2, 8, 1, 9};
    
    // Find maximum
    int max = arr[0];
    for (int i = 1; i < 5; i++) {
        if (arr[i] > max) {
            max = arr[i];
        }
    }
    printf("\nMaximum value: %d\n", max);
    
    // Find minimum

    for (int i = 1; i < 5; i++) {
        if (arr[i] < min) {
            min = arr[i];
        }

    int sum = 0;
    for (int i = 0; i < 5; i++) {
        sum += arr[i];

Functions & Pointers

Functions

Defining and using functions in C

#include <stdio.h>

// Function declarations (prototypes)
void greet();
int add(int a, int b);
float calculateAverage(int arr[], int size);
void swap(int *a, int *b);
int factorial(int n);
int isPrime(int num);

int main() {
    // Basic function call
    greet();
    
    // Function with parameters and return value
    int result = add(5, 3);
    printf("5 + 3 = %d\n", result);
    
    // Function with array parameter
    int numbers[] = {10, 20, 30, 40, 50};
    int size = sizeof(numbers) / sizeof(numbers[0]);
    float avg = calculateAverage(numbers, size);
    printf("Average: %.2f\n", avg);
    
    // Function with pointers (call by reference)
    int x = 10, y = 20;
    printf("Before swap: x = %d, y = %d\n", x, y);
    swap(&x, &y);
    printf("After swap: x = %d, y = %d\n", x, y);
    
    // Recursive function
    int num = 5;
    printf("Factorial of %d is %d\n", num, factorial(num));
    
    // Function returning boolean-like value
    int testNum = 17;
    if (isPrime(testNum)) {
        printf("%d is a prime number\n", testNum);
    } else {
        printf("%d is not a prime number\n", testNum);
    }
    
    return 0;
}

// Function definitions

// Basic function without parameters
void greet() {
    printf("Hello from greet() function!\n");
}

// Function with parameters and return value
int add(int a, int b) {
    return a + b;
}

// Function with array parameter
float calculateAverage(int arr[], int size) {
    int sum = 0;
    for (int i = 0; i < size; i++) {
        sum += arr[i];
    }
    return (float)sum / size;
}

// Function using pointers (call by reference)
void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

// Recursive function
int factorial(int n) {
    if (n == 0 || n == 1) {
        return 1;
    }
    return n * factorial(n - 1);
}

// Function returning int (boolean-like)
int isPrime(int num) {
    if (num <= 1) return 0;
    if (num == 2) return 1;
    if (num % 2 == 0) return 0;
    
    for (int i = 3; i * i <= num; i += 2) {
        if (num % i == 0) {
            return 0;
        }
    }
    return 1;
}

// Function with static variable
int counter() {
    static int count = 0;  // Retains value between calls
    count++;
    return count;
}

// Example using static variable
void staticExample() {
    printf("\nStatic variable example:\n");
    for (int i = 0; i < 5; i++) {
        printf("Counter: %d\n", counter());
    }
}

#include <stdio.h>

int main() {
    // Basic pointers
    int x = 10;
    int *p = &x;  // p points to x
    
    printf("Basic Pointers:\n");
    printf("Value of x: %d\n", x);
    printf("Address of x: %p\n", (void*)&x);
    printf("Value of p (address stored): %p\n", (void*)p);
    printf("Value at address stored in p: %d\n", *p);
    
    // Pointer arithmetic
    printf("\nPointer Arithmetic:\n");
    int arr[] = {10, 20, 30, 40, 50};
    int *ptr = arr;  // Points to first element
    
    printf("Array elements using pointers:\n");
    for (int i = 0; i < 5; i++) {
        printf("arr[%d] = %d, *(ptr + %d) = %d\n", 
               i, arr[i], i, *(ptr + i));
    }
    
    // Array name as pointer
    printf("\nArray name as pointer:\n");
    printf("*arr = %d\n", *arr);
    printf("*(arr + 1) = %d\n", *(arr + 1));
    printf("*(arr + 2) = %d\n", *(arr + 2));
    
    // Pointer to pointer
    printf("\nPointer to Pointer:\n");
    int y = 100;
    int *p1 = &y;
    int **p2 = &p1;
    
    printf("y = %d\n", y);
    printf("*p1 = %d\n", *p1);
    printf("**p2 = %d\n", **p2);
    printf("Address of y: %p\n", (void*)&y);
    printf("Address stored in p1: %p\n", (void*)p1);
    printf("Address stored in p2: %p\n", (void*)p2);
    
    // Pointers and functions
    printf("\nPointers and Functions:\n");
    int a = 5, b = 10;
    printf("Before: a = %d, b = %d\n", a, b);
    
    // Swap using pointers
    void swap(int *x, int *y) {

        *x = *y;
        *y = temp;

    printf("After swap: a = %d, b = %d\n", a, b);
    
    // Pointers with arrays in functions

        printf("Array elements: ");
        for (int i = 0; i < size; i++) {
            printf("%d ", *(arr + i));
        }
        printf("\n");

    printArray(arr, 5);
    
    // Pointer and string
    printf("\nPointers and Strings:\n");
    char str[] = "Hello";

    printf("String: %s\n", str);
    printf("Using pointer: ");
    while (*strPtr != '\0') {
        printf("%c", *strPtr);
        strPtr++;
    }
    printf("\n");
    
    // Array of pointers
    printf("\nArray of Pointers:\n");
    char *names[] = {"Alice", "Bob", "Charlie", "Diana"};
    int namesCount = sizeof(names) / sizeof(names[0]);
    
    for (int i = 0; i < namesCount; i++) {
        printf("Name %d: %s\n", i + 1, names[i]);
    }
    
    // Void pointer
    printf("\nVoid Pointer:\n");
    int num = 42;
    float fnum = 3.14;
    char ch = 'A';
    
    void *vp;
    
    vp = #
    printf("Integer via void pointer: %d\n", *(int*)vp);

    vp = &ch;

Structures

Creating and using structures for complex data

#include <stdio.h>
#include <string.h>

// Structure definition
struct Student {
    char name[50];
    int age;
    float gpa;
    int roll_number;
};

// Structure with typedef
typedef struct {
    char title[100];
    char author[50];
    int year;
    float price;
} Book;

// Nested structure
struct Address {
    char street[100];
    char city[50];
    char state[50];
    int zip_code;
};

struct Person {
    char name[50];
    int age;
    struct Address address;
};

// Function to print student information
void printStudent(struct Student s) {
    printf("Name: %s\n", s.name);
    printf("Age: %d\n", s.age);
    printf("GPA: %.2f\n", s.gpa);
    printf("Roll Number: %d\n", s.roll_number);
}

// Function to update GPA (using pointer)
void updateGPA(struct Student *s, float newGPA) {
    s->gpa = newGPA;
}

int main() {
    // Structure variable declaration and initialization
    struct Student student1 = {"Alice Johnson", 20, 3.8, 101};
    
    printf("Student 1 Information:\n");
    printStudent(student1);
    
    // Modifying structure members
    printf("\nAfter modification:\n");
    student1.age = 21;
    student1.gpa = 3.9;
    strcpy(student1.name, "Alice Smith");
    printStudent(student1);
    
    // Using typedef structure
    Book book1 = {"The C Programming Language", "K&R", 1978, 45.99};
    
    printf("\nBook Information:\n");
    printf("Title: %s\n", book1.title);
    printf("Author: %s\n", book1.author);
    printf("Year: %d\n", book1.year);
    printf("Price: $%.2f\n", book1.price);
    
    // Array of structures
    struct Student class[3] = {
        {"Bob Brown", 19, 3.5, 102},
        {"Charlie Davis", 20, 3.2, 103},
        {"Diana Evans", 21, 3.9, 104}
    };
    
    printf("\nClass Students:\n");
    for (int i = 0; i < 3; i++) {
        printf("Student %d: %s (GPA: %.2f)\n", 
               class[i].roll_number, class[i].name, class[i].gpa);
    }
    
    // Nested structures
    struct Person person1 = {
        "John Doe",
        30,
        {"123 Main St", "Springfield", "IL", 62701}
    };
    
    printf("\nPerson with Address:\n");
    printf("Name: %s\n", person1.name);
    printf("Age: %d\n", person1.age);
    printf("Address: %s, %s, %s %d\n", 
           person1.address.street, 
           person1.address.city, 
           person1.address.state, 
           person1.address.zip_code);
    
    // Pointers to structures
    struct Student *ptr = &student1;
    
    printf("\nUsing Structure Pointer:\n");
    printf("Name: %s\n", ptr->name);        // Arrow operator
    printf("Age: %d\n", ptr->age);
    printf("GPA: %.2f\n", (*ptr).gpa);      // Dot operator with dereference
    
    // Update using pointer
    updateGPA(&student1, 4.0);
    printf("\nAfter GPA update:\n");
    printf("New GPA: %.2f\n", student1.gpa);
    
    // Structure with array
    struct Course {
        char name[50];
        int credits;
        float grades[5];
    };
    
    struct Course math = {"Mathematics", 3, {85.5, 90.0, 78.5, 92.0, 88.5}};
    
    printf("\nCourse Information:\n");
    printf("Course: %s (%d credits)\n", math.name, math.credits);
    printf("Grades: ");
    for (int i = 0; i < 5; i++) {
        printf("%.1f ", math.grades[i]);
    }
    printf("\n");
    
    // Structure size
    printf("\nStructure Sizes:\n");
    printf("Size of Student: %zu bytes\n", sizeof(struct Student));
    printf("Size of Book: %zu bytes\n", sizeof(Book));
    printf("Size of Person: %zu bytes\n", sizeof(struct Person));
    
    return 0;
}

Reading from and writing to files

#include <stdio.h>

int main() {
    FILE *file;

    // Writing to a file
    printf("Writing to file...\n");
    file = fopen("example.txt", "w");

Dynamic Memory Allocation

Managing memory at runtime with malloc, calloc, realloc, free

#include <stdio.h>
#include <stdlib.h>

int main() {
    // malloc - Memory Allocation
    printf("=== malloc Example ===\n");
    int *mallocPtr = (int*)malloc(5 * sizeof(int));
    
    if (mallocPtr == NULL) {
        printf("Memory allocation failed!\n");
        return 1;
    }
    
    // Initialize and use allocated memory
    for (int i = 0; i < 5; i++) {
        mallocPtr[i] = (i + 1) * 10;
    }
    
    printf("malloc array: ");
    for (int i = 0; i < 5; i++) {
        printf("%d ", mallocPtr[i]);
    }
    printf("\n");
    
    // calloc - Contiguous Allocation (initialized to 0)
    printf("\n=== calloc Example ===\n");
    int *callocPtr = (int*)calloc(5, sizeof(int));
    
    if (callocPtr == NULL) {
        printf("Memory allocation failed!\n");
        free(mallocPtr);
        return 1;
    }
    
    printf("calloc array (initialized to 0): ");
    for (int i = 0; i < 5; i++) {
        printf("%d ", callocPtr[i]);
    }
    printf("\n");
    
    // Initialize calloc array
    for (int i = 0; i < 5; i++) {
        callocPtr[i] = (i + 1) * 20;
    }
    
    printf("calloc array after assignment: ");
    for (int i = 0; i < 5; i++) {
        printf("%d ", callocPtr[i]);
    }
    printf("\n");
    
    // realloc - Reallocate memory
    printf("\n=== realloc Example ===\n");
    int *reallocPtr = (int*)realloc(callocPtr, 10 * sizeof(int));
    
    if (reallocPtr == NULL) {
        printf("Memory reallocation failed!\n");
        free(mallocPtr);
        free(callocPtr);
        return 1;
    }
    
    callocPtr = reallocPtr;  // Use the new pointer
    
    // Initialize new elements
    for (int i = 5; i < 10; i++) {
        callocPtr[i] = (i + 1) * 20;
    }
    
    printf("After realloc (size 10): ");
    for (int i = 0; i < 10; i++) {
        printf("%d ", callocPtr[i]);
    }
    printf("\n");
    
    // Dynamic string allocation
    printf("\n=== Dynamic Strings ===\n");
    char *str = (char*)malloc(50 * sizeof(char));
    
    if (str != NULL) {
        strcpy(str, "Hello, Dynamic Memory!");
        printf("Dynamic string: %s\n", str);
    }
    
    // Dynamic array of structures
    printf("\n=== Dynamic Structure Array ===\n");
    typedef struct {
        char name[50];
        int age;
    } Person;
    
    int numPersons = 3;
    Person *persons = (Person*)malloc(numPersons * sizeof(Person));
    
    if (persons != NULL) {
        // Initialize structures
        strcpy(persons[0].name, "Alice");
        persons[0].age = 25;
        
        strcpy(persons[1].name, "Bob");
        persons[1].age = 30;
        
        strcpy(persons[2].name, "Charlie");
        persons[2].age = 35;
        
        // Print structures
        for (int i = 0; i < numPersons; i++) {
            printf("Person %d: %s, %d years\n", 
                   i + 1, persons[i].name, persons[i].age);
        }
    }
    
    // Two-dimensional dynamic array
    printf("\n=== 2D Dynamic Array ===\n");
    int rows = 3, cols = 4;
    int **matrix = (int**)malloc(rows * sizeof(int*));
    
    for (int i = 0; i < rows; i++) {
        matrix[i] = (int*)malloc(cols * sizeof(int));
    }
    
    // Initialize matrix
    int counter = 1;
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            matrix[i][j] = counter++;
        }
    }
    
    // Print matrix
    printf("Dynamic 2D Array:\n");
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            printf("%2d ", matrix[i][j]);
        }
        printf("\n");
    }
    
    // Free all allocated memory
    printf("\n=== Freeing Memory ===\n");
    free(mallocPtr);
    free(callocPtr);
    free(str);
    free(persons);
    
    // Free 2D array
    for (int i = 0; i < rows; i++) {
        free(matrix[i]);
    }
    free(matrix);
    
    printf("All memory freed successfully!\n");
    
    // Memory leak demonstration (commented out for safety)
    /*
    printf("\n=== Memory Leak Example ===\n");
    int *leakPtr = (int*)malloc(100 * sizeof(int));
    // Forgot to free leakPtr - memory leak!
    // free(leakPtr);  // This should be called
    */
    
    return 0;
}

Recursion

Functions that call themselves

#include <stdio.h>

// Factorial function
unsigned long long factorial(int n) {
    // Base case
    if (n == 0 || n == 1) {
        return 1;
    }
    // Recursive case
    return n * factorial(n - 1);
}

// Fibonacci sequence
int fibonacci(int n) {
    // Base cases
    if (n == 0) return 0;
    if (n == 1) return 1;
    // Recursive case
    return fibonacci(n - 1) + fibonacci(n - 2);
}

// Power calculation
double power(double base, int exponent) {
    // Base case
    if (exponent == 0) return 1;
    if (exponent == 1) return base;
    
    // Recursive cases
    if (exponent > 0) {
        return base * power(base, exponent - 1);
    } else {
        return 1 / power(base, -exponent);
    }
}

// GCD (Greatest Common Divisor) using Euclidean algorithm
int gcd(int a, int b) {
    // Base case
    if (b == 0) return a;
    // Recursive case
    return gcd(b, a % b);
}

// Tower of Hanoi
void towerOfHanoi(int n, char from_rod, char to_rod, char aux_rod) {
    if (n == 1) {
        printf("Move disk 1 from %c to %c\n", from_rod, to_rod);
        return;
    }
    towerOfHanoi(n - 1, from_rod, aux_rod, to_rod);
    printf("Move disk %d from %c to %c\n", n, from_rod, to_rod);
    towerOfHanoi(n - 1, aux_rod, to_rod, from_rod);
}

// Binary search using recursion
int binarySearch(int arr[], int left, int right, int target) {
    if (left <= right) {
        int mid = left + (right - left) / 2;
        
        if (arr[mid] == target) return mid;
        if (arr[mid] > target) return binarySearch(arr, left, mid - 1, target);
        return binarySearch(arr, mid + 1, right, target);
    }
    return -1;  // Not found
}

// Sum of digits
int sumOfDigits(int n) {
    if (n == 0) return 0;
    return (n % 10) + sumOfDigits(n / 10);
}

// Reverse a number
int reverseNumber(int n, int reversed) {
    if (n == 0) return reversed;
    return reverseNumber(n / 10, reversed * 10 + n % 10);
}

// Palindrome check
int isPalindrome(char str[], int start, int end) {
    // Base case: single character or empty string
    if (start >= end) return 1;
    
    // If characters don't match
    if (str[start] != str[end]) return 0;
    
    // Recursive call for remaining substring
    return isPalindrome(str, start + 1, end - 1);
}

    printf("GCD(17, 13) = %d\n", gcd(17, 13));
    
    // Tower of Hanoi
    printf("\nTower of Hanoi for 3 disks:\n");
    towerOfHanoi(3, 'A', 'C', 'B');
    
    // Binary Search
    printf("\nBinary Search:\n");
    int sortedArray[] = {2, 5, 8, 12, 16, 23, 38, 45, 67, 89};
    int size = sizeof(sortedArray) / sizeof(sortedArray[0]);
    int target = 23;
    
    int result = binarySearch(sortedArray, 0, size - 1, target);
    if (result != -1) {

        if (n <= 0) return 0;

    int arrayMax(int a[], int n) {
        if (n == 1) return a[0];
        int max = arrayMax(a, n - 1);
        return (a[n - 1] > max) ? a[n - 1] : max;
    }
    
    printf("Array sum: %d\n", arraySum(arr, arrSize));
    printf("Array max: %d\n", arrayMax(arr, arrSize));
    
    return 0;
}

Interview Topics

Important C concepts for technical interviews

📚 Storage Classes

auto - Default, local scope
register - Store in CPU register
static - Preserves value between calls
extern - Global variable declaration

🔄 Call by Value vs Reference

// Call by value - copies value
void modifyValue(int x) {
    x = x + 10;  // Doesn't affect original
}

// Call by reference - uses pointers
void modifyReference(int *x) {
    *x = *x + 10;  // Affects original
}

⚠️ Dangling Pointers

int *createDanglingPointer() {
    int x = 10;
    return &x;  // x goes out of scope - dangling pointer!
}

// Safe alternative
int *createSafePointer() {
    int *x = malloc(sizeof(int));
    *x = 10;
    return x;  // Caller must free this memory
}

💾 malloc vs calloc

malloc - Allocates memory, contains garbage values
calloc - Allocates and initializes to zero
realloc - Resizes allocated memory
free - Deallocates memory

🏗️ Stack vs Heap

Stack - Automatic memory, fast, limited size
Heap - Dynamic memory, manual management, larger
Use stack for local variables, function calls
Use heap for dynamic data, large objects
