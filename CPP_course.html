    cout << "x || y: " << (x || y) << endl;
    cout << "!x: " << (!x) << endl;
    
    // Assignment operators
    int c = 10;
    cout << "\n--- Assignment Operators ---" << endl;
    cout << "Initial c: " << c << endl;
    c += 5;  cout << "c += 5: " << c << endl;
    c -= 3;  cout << "c -= 3: " << c << endl;
    c *= 2;  cout << "c *= 2: " << c << endl;
    c /= 4;  cout << "c /= 4: " << c << endl;
    c %= 3;  cout << "c %= 3: " << c << endl;
    
    // Increment/Decrement operators
    int count = 5;
    cout << "\n--- Increment/Decrement ---" << endl;
    cout << "count: " << count << endl;
    cout << "count++: " << count++ << endl;  // Post-increment
    cout << "count: " << count << endl;
    cout << "++count: " << ++count << endl;  // Pre-increment
    cout << "count--: " << count-- << endl;  // Post-decrement
    cout << "--count: " << --count << endl;  // Pre-decrement
    
    // Bitwise operators
    int num1 = 5, num2 = 3;  // 5 = 101, 3 = 011
    cout << "\n--- Bitwise Operators ---" << endl;
    cout << num1 << " & " << num2 << " = " << (num1 & num2) << endl;  // AND
    cout << num1 << " | " << num2 << " = " << (num1 | num2) << endl;  // OR
    cout << num1 << " ^ " << num2 << " = " << (num1 ^ num2) << endl;  // XOR
    cout << "~" << num1 << " = " << (~num1) << endl;  // NOT
    cout << num1 << " << 1 = " << (num1 << 1) << endl;  // Left shift
    cout << num1 << " >> 1 = " << (num1 >> 1) << endl;  // Right shift
    
    return 0;
}

Loops

For loops, while loops, and loop control

#include <iostream>
using namespace std;

int main() {
    // For loop
    cout << "--- For Loop ---" << endl;
    cout << "Counting from 1 to 5:" << endl;
    for (int i = 1; i <= 5; i++) {
        cout << i << " ";
    }
    cout << endl;
    
    // For loop with array
    cout << "\n--- For Loop with Array ---" << endl;
    int numbers[] = {10, 20, 30, 40, 50};
    cout << "Array elements: ";
    for (int i = 0; i < 5; i++) {
        cout << numbers[i] << " ";
    }
    cout << endl;
    
    // Range-based for loop (C++11)
    cout << "\n--- Range-based For Loop ---" << endl;
    cout << "Array elements: ";
    for (int num : numbers) {
        cout << num << " ";
    }
    cout << endl;
    
    // While loop
    cout << "\n--- While Loop ---" << endl;
    cout << "Countdown:" << endl;
    int count = 5;
    while (count > 0) {
        cout << count << " ";
        count--;
    }
    cout << "Blast off!" << endl;
    
    // Do-while loop
    cout << "\n--- Do-While Loop ---" << endl;
    int input;
    do {
        cout << "Enter a positive number: ";
        cin >> input;
    } while (input <= 0);
    cout << "You entered: " << input << endl;
    
    // Nested loops
    cout << "\n--- Nested Loops ---" << endl;
    cout << "Multiplication table (1-3):" << endl;
    for (int i = 1; i <= 3; i++) {
        for (int j = 1; j <= 3; j++) {
            cout << i << " x " << j << " = " << (i * j) << "\t";
        }
        cout << endl;
    }
    
    // Loop control: break and continue
    cout << "\n--- Loop Control ---" << endl;
    cout << "Numbers 1-10 (skip 5):" << endl;
    for (int i = 1; i <= 10; i++) {
        if (i == 5) {
            continue;  // Skip 5
        }
        if (i == 8) {
            break;     // Stop at 8
        }
        cout << i << " ";
    }
    cout << endl;
    
    // Infinite loop with break
    cout << "\n--- Infinite Loop with Break ---" << endl;
    int sum = 0, number;
    while (true) {
        cout << "Enter a number (0 to stop): ";
        cin >> number;
        if (number == 0) {
            break;
        }
        sum += number;
    }
    cout << "Total sum: " << sum << endl;
    
    // Loop with multiple variables
    cout << "\n--- Loop with Multiple Variables ---" << endl;
    for (int i = 1, j = 10; i <= 10; i++, j--) {
        cout << "i=" << i << ", j=" << j << endl;
    }
    
    return 0;
}

            min = arr[i];
        }

    int sum = 0;
    for (int num : arr) {
        sum += num;
    }
    cout << "Sum: " << sum << endl;
    cout << "Average: " << (double)sum / arrSize << endl;
    
    return 0;

#include <iostream>

}

    const double PI = 3.14159;
    return PI * radius * radius;
}

// Function with array parameter

    cout << "Array elements: ";
    for (int i = 0; i < size; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
}

// Recursive function
int factorial(int n) {
    if (n == 0 || n == 1) {
        return 1;
    }
    return n * factorial(n - 1);
}

// Function overloading
int multiply(int a, int b) {
    return a * b;
}

double multiply(double a, double b) {
    return a * b;
}

int multiply(int a, int b, int c) {
    return a * b * c;
}

int sum(int a, int b, int c) {
    return a + b + c;
}

void incrementByValue(int x) {

    cout << "Inside function (by value): " << x << endl;
}

// Pass by reference
void incrementByReference(int &x) {
    x++;
    cout << "Inside function (by reference): " << x << endl;
}

// Pass by pointer
void incrementByPointer(int *x) {
    (*x)++;
    cout << "Inside function (by pointer): " << *x << endl;
}

// Inline function
inline int square(int x) {
    return x * x;
}

// Class with constructor
class Car {
public:
    string brand;
    string model;
    int year;
    double price;
    
    // Default constructor
    Car() {
        brand = "Unknown";
        model = "Unknown";
        year = 0;
        price = 0.0;
        cout << "Default constructor called!" << endl;
    }
    
    // Parameterized constructor
    Car(string b, string m, int y, double p) {
        brand = b;
        model = m;
        year = y;
        price = p;
        cout << "Parameterized constructor called!" << endl;
    }
    
    // Copy constructor
    Car(const Car &other) {
        brand = other.brand;
        model = other.model;
        year = other.year;
        price = other.price;
        cout << "Copy constructor called!" << endl;
    }
    
    // Destructor
    ~Car() {
        cout << "Destructor called for " << brand << " " << model << endl;
    }
    
    // Member function
    void displayInfo() {
        cout << year << " " << brand << " " << model << " - $" << price << endl;
    }
    
    // Setter method
    void setPrice(double p) {
        if (p >= 0) {
            price = p;
        } else {
            cout << "Invalid price!" << endl;
        }
    }
    
    // Getter method
    double getPrice() {
        return price;
    }
};

class BankAccount {

    // Private data members (encapsulation)
    string accountNumber;
    string accountHolder;
    double balance;
    string password;
    
    // Private helper method
    bool authenticate(string inputPassword) {
        return this->password == inputPassword;
    }
    
public:
    // Constructor
    BankAccount(string accNum, string holder, double initialBalance, string pwd) {
        accountNumber = accNum;
        accountHolder = holder;
        balance = initialBalance;
        password = pwd;
    }
    
    // Public getters (accessor methods)
    string getAccountNumber() const {
        return accountNumber;
    }
    
    string getAccountHolder() const {
        return accountHolder;
    }
    
    double getBalance() const {
        return balance;
    }
    
    // No setter for account number (immutable)
    
    // Public methods for operations
    void deposit(double amount) {
        if (amount > 0) {
            balance += amount;
            cout << "Deposited: $" << amount << endl;
            cout << "New balance: $" << balance << endl;
        } else {
            cout << "Invalid deposit amount!" << endl;
        }
    }
    
    void withdraw(double amount, string inputPassword) {
        if (!authenticate(inputPassword)) {
            cout << "Invalid password! Withdrawal failed." << endl;
            return;
        }
        
        if (amount > 0 && amount <= balance) {
            balance -= amount;
            cout << "Withdrawn: $" << amount << endl;
            cout << "New balance: $" << balance << endl;
        } else {
            cout << "Invalid withdrawal amount or insufficient funds!" << endl;
        }
    }
    
    void transfer(BankAccount &recipient, double amount, string inputPassword) {
        if (!authenticate(inputPassword)) {
            cout << "Invalid password! Transfer failed." << endl;
            return;
        }
        
        if (amount > 0 && amount <= balance) {
            balance -= amount;
            recipient.balance += amount;  // Direct access to private member (same class)
            cout << "Transferred: $" << amount << " to " << recipient.accountHolder << endl;
        } else {
            cout << "Invalid transfer amount or insufficient funds!" << endl;
        }
    }
    
    // Public method to change password
    void changePassword(string oldPassword, string newPassword) {
        if (authenticate(oldPassword)) {
            password = newPassword;
            cout << "Password changed successfully!" << endl;
        } else {
            cout << "Invalid old password!" << endl;
        }
    }
    
    // Display account info (without password)
    void displayAccountInfo() const {
        cout << "Account Number: " << accountNumber << endl;
        cout << "Account Holder: " << accountHolder << endl;
        cout << "Balance: $" << balance << endl;
    }
};

void exceptionSafety() {
    cout << "\n=== EXCEPTION SAFETY ===" << endl;
    
    class Database {
    private:
        vector data;
        
    public:
        // Basic guarantee - state remains valid
        void addDataBasic(const string& item) {
            data.push_back(item);
            if (item == "throw") {
                throw runtime_error("Simulated error");
            }
        }
        
        // Strong guarantee - commit or rollback
        void addDataStrong(const string& item) {
            vector oldData = data;  // Save state
            data.push_back(item);
            
            if (item == "throw") {
                data = oldData;  // Rollback
                throw runtime_error("Simulated error - rolled back");
            }
        }
        
        // Noexcept guarantee - never throws
        size_t size() const noexcept {
            return data.size();
        }
        
        void print() const {
            cout << "Database contents: ";
            for (const auto& item : data) {
                cout << item << " ";
            }
            cout << endl;
        }
    };
    
    Database db;
    
    // Test basic guarantee
    try {
        db.addDataBasic("item1");
        db.addDataBasic("throw");
    } catch (const exception& e) {
        cout << "Basic guarantee - caught: " << e.what() << endl;
        db.print();  // State might be modified
    }
    
    // Test strong guarantee
    try {
        db.addDataStrong("item2");
        db.addDataStrong("throw");
    } catch (const exception& e) {
        cout << "Strong guarantee - caught: " << e.what() << endl;
        db.print();  // State should be unchanged
    }
}

        if (student) {
            student->updateMarks(newMarks);
            cout << "Marks updated successfully!" << endl;
        } else {
            cout << "Student not found!" << endl;
        }
    }
    
    void deleteStudent(int rollNumber) {
        auto it = remove_if(students.begin(), students.end(),
            [rollNumber](const Student& s) {
                return s.getRollNumber() == rollNumber;
            });
        
        if (it != students.end()) {
            students.erase(it, students.end());
            cout << "Student deleted successfully!" << endl;
        } else {
            cout << "Student not found!" << endl;
        }
    }
    
    void displayTopStudents(int count = 3) const {
        if (students.empty()) {
            cout << "No students in the system." << endl;
            return;
        }
        
        vector sortedStudents = students;
        sort(sortedStudents.begin(), sortedStudents.end(),
            [](const Student& a, const Student& b) {
                return a.getMarks() > b.getMarks();
            });
        
        cout << "\n=== TOP " << count << " STUDENTS ===" << endl;
        for (int i = 0; i < min(count, static_cast(sortedStudents.size())); i++) {
            sortedStudents[i].display();
        }
    }
};

class Calculator {
private:
    vector history;
    
public:
    double add(double a, double b) {
        double result = a + b;
        history.push_back(to_string(a) + " + " + to_string(b) + " = " + to_string(result));
        return result;
    }
    
    double subtract(double a, double b) {
        double result = a - b;
        history.push_back(to_string(a) + " - " + to_string(b) + " = " + to_string(result));
        return result;
    }
    
    double multiply(double a, double b) {
        double result = a * b;
        history.push_back(to_string(a) + " * " + to_string(b) + " = " + to_string(result));
        return result;
    }
    
    double divide(double a, double b) {
        if (b == 0) {
            throw runtime_error("Division by zero!");
        }
        double result = a / b;
        history.push_back(to_string(a) + " / " + to_string(b) + " = " + to_string(result));
        return result;
    }
    
    void showHistory() const {
        if (history.empty()) {
            cout << "No calculation history." << endl;
            return;
        }
        
        cout << "\n=== CALCULATION HISTORY ===" << endl;
        for (size_t i = 0; i < history.size(); i++) {
            cout << i + 1 << ". " << history[i] << endl;
        }
    }
    
    void clearHistory() {
        history.clear();
        cout << "History cleared!" << endl;
    }
};
